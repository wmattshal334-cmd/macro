    TargetPlayer = "Auto (Closest)",

local WEAPONS = { "golden", "normal", "rpg" } -- Golden + Normal Snowball Launchers, and the RPG (slot 5)

-- ---------- projectile calibration (learned from the real projectiles) ----------
local CAL_FILE = "Exotica_Aim_Calibration.json"
local Projectile = { speed = {}, drop = {}, n = {}, onUpdate = nil }
for _, w in ipairs(WEAPONS) do
    Projectile.speed[w] = 120 -- placeholder until the first real shot is measured
    Projectile.drop[w] = 0
    Projectile.n[w] = 0
end

if readfile and isfile and isfile(CAL_FILE) then
    pcall(function()
        local d = HttpService:JSONDecode(readfile(CAL_FILE))
        for _, w in ipairs(WEAPONS) do
            local e = d[w]
            if type(e) == "table" and type(e.speed) == "number" then
                Projectile.speed[w] = e.speed
                Projectile.drop[w] = tonumber(e.drop) or 0
                Projectile.n[w] = math.min(tonumber(e.n) or 0, 5)
            end
        end
    end)
end

local saveQueued = false
local function saveCalibration()
    if not writefile or saveQueued then return end
    saveQueued = true
    task.delay(2, function()
        saveQueued = false
        pcall(function()
            local d = {}
            for _, w in ipairs(WEAPONS) do
                d[w] = { speed = Projectile.speed[w], drop = Projectile.drop[w], n = Projectile.n[w] }
            end
            writefile(CAL_FILE, HttpService:JSONEncode(d))
        end)
    end)
end

local function updateEstimate(weapon, speed, drop)
    if speed < 15 or speed > 1500 then return end
    local n = Projectile.n[weapon] or 0
    local cur = Projectile.speed[weapon]
    if n >= 3 and (speed < cur * 0.5 or speed > cur * 2) then return end -- outlier (wrong part matched)
    if n == 0 then
        Projectile.speed[weapon] = speed
    else
        Projectile.speed[weapon] = cur + (speed - cur) * 0.3
    end
    if drop and math.abs(drop) < 400 then
        Projectile.drop[weapon] = (n == 0) and drop or (Projectile.drop[weapon] + (drop - Projectile.drop[weapon]) * 0.2)
    end
    Projectile.n[weapon] = n + 1
    saveCalibration()
    if Projectile.onUpdate then pcall(Projectile.onUpdate) end
end

local function currentSpeed(weapon)
    if not AssistSettings.SpeedAuto then return AssistSettings.ManualSpeed end
    return Projectile.speed[weapon] or 120
end

local function myRoot()
    local c = p.Character
    return c and c:FindFirstChild("HumanoidRootPart")
end

-- Every aimed click is logged. The next new small part that appears near you within
-- 0.8s is matched to the oldest unmatched shot, then its real speed and drop are measured.
local pendingShots = {}
local trackedGroups = setmetatable({}, { __mode = "k" })

local function registerShot(weapon)
    if table.find(WEAPONS, weapon) then
        table.insert(pendingShots, { weapon = weapon, t = clock() })
    end
end

local function isCharacterPart(inst)
    for _, plr in ipairs(Players:GetPlayers()) do
        local c = plr.Character
        if c and inst:IsDescendantOf(c) then return true end
    end
    return false
end

local function trackProjectile(part, weapon)
    task.spawn(function()
        local samples = {}
        local startT = clock()
        while part.Parent and clock() - startT < 0.45 do
            table.insert(samples, { t = clock(), p = part.Position })
            RunService.Heartbeat:Wait()
        end
        if #samples < 4 then return end
        local a, b = samples[2], samples[#samples] -- skip the spawn frame
        local dt = b.t - a.t
        if dt < 0.08 then return end
        local speed = (b.p - a.p).Magnitude / dt

        local drop = nil
        if #samples >= 7 then
            local m = samples[math.floor((#samples + 2) / 2)]
            if m.t > a.t and b.t > m.t then
                local vy1 = (m.p.Y - a.p.Y) / (m.t - a.t)
                local vy2 = (b.p.Y - m.p.Y) / (b.t - m.t)
                drop = -((vy2 - vy1) / ((b.t - a.t) / 2)) -- positive = falls
            end
        end
        updateEstimate(weapon, speed, drop)
    end)
end

workspace.DescendantAdded:Connect(function(inst)
    if #pendingShots == 0 or not inst:IsA("BasePart") then return end
    local now = clock()
    while pendingShots[1] and now - pendingShots[1].t > 0.8 do table.remove(pendingShots, 1) end
    local shot = pendingShots[1]
    local root = myRoot()
    if not (shot and root) then return end
    if inst.Size.Magnitude > 6 then return end
    if (inst.Position - root.Position).Magnitude > 25 then return end
    if isCharacterPart(inst) then return end
    local group = inst:FindFirstAncestorOfClass("Model") or inst
    if trackedGroups[group] then return end
    trackedGroups[group] = true
    table.remove(pendingShots, 1)
    trackProjectile(inst, shot.weapon)
end)

-- ---------- target tracking (runs every frame, even with Assist off) ----------
local TargetState = {}
local lockedPlr = nil

local function updateTarget(plr, now)
    local char = plr.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not (root and hum and hum.Health > 0) then
        TargetState[plr] = nil
        return
    end
    local st = TargetState[plr]
    if not st or st.root ~= root then
        st = { root = root, hum = hum, hist = {}, vel = Vector3.zero, air = false }
        TargetState[plr] = st
    end
    local hist, pos = st.hist, root.Position
    local newest = hist[#hist]
    if newest and (pos - newest.p).Magnitude > 40 then -- teleport / respawn, drop history
        table.clear(hist)
        st.vel = Vector3.zero
    end
    table.insert(hist, { t = now, p = pos })
    while #hist > 2 and now - hist[1].t > 0.3 do table.remove(hist, 1) end

    local ref = hist[1]
    for i = 1, #hist do
        if now - hist[i].t <= 0.16 then ref = hist[i] break end
    end
    local dt = now - ref.t
    if dt > 0.02 then
        local measured = (pos - ref.p) / dt
        local blended = measured * 0.6 + root.AssemblyLinearVelocity * 0.4
        st.vel = st.vel:Lerp(blended, 0.6)
    end
    st.air = hum.FloorMaterial == Enum.Material.Air
end

local function toScreen(world)
    local cam = workspace.CurrentCamera
    local sp, on = cam:WorldToViewportPoint(world)
    if on and sp.Z > 0 then return Vector2.new(sp.X, sp.Y) end
    return nil
end

local function oneWayLatency()
    if not AssistSettings.LatencyComp then return 0 end
    local ok, ping = pcall(function() return p:GetNetworkPing() end)
    if not ok or type(ping) ~= "number" then return 0 end
    return math.clamp(ping * 0.5, 0, 0.3)
end

local function groundBelow(from, ignore)
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = ignore
    local r = workspace:Raycast(from, Vector3.new(0, -300, 0), params)
    return r and r.Position.Y
end

-- Iterative intercept: guess flight time from distance, predict where the target will be
-- at that time, recompute the distance to THAT point, repeat. Converges in a few passes.
-- Grounded targets: horizontal velocity only. Airborne: real gravity, and the landing
-- point is clamped to the floor so it doesn't aim under the map.
local function predictFor(plr, weapon)
    local st = TargetState[plr]
    local myR = myRoot()
    if not (st and myR) then return nil end
    local root = st.root
    local origin = myR.Position
    local speed = math.max(currentSpeed(weapon), 10)
    local pos, vel = root.Position, st.vel
    local acc = Vector3.zero
    if st.air then
        acc = Vector3.new(0, -workspace.Gravity, 0)
    else
        vel = Vector3.new(vel.X, 0, vel.Z)
    end

    local extra, lead = oneWayLatency(), AssistSettings.Lead
    local pred, flight = pos, (pos - origin).Magnitude / speed
    for _ = 1, 6 do
        local tt = math.min((flight + extra) * lead, 3)
        pred = pos + vel * tt + acc * (0.5 * tt * tt)
        flight = (pred - origin).Magnitude / speed
    end

    if st.air then
        local off = math.max(st.hum.HipHeight + root.Size.Y * 0.5, 3)
        local from = Vector3.new(pred.X, math.max(pos.Y, pred.Y) + 2, pred.Z)
        local gy = groundBelow(from, { p.Character, plr.Character })
        if gy and pred.Y < gy + off then
            pred = Vector3.new(pred.X, gy + off, pred.Z)
        end
    end

    if AssistSettings.ArcComp and (Projectile.n[weapon] or 0) >= 3 then
        local d = Projectile.drop[weapon] or 0
        if d > 1 then
            local up = math.min(0.5 * d * flight * flight, (pred - origin).Magnitude * 0.5)
            pred = pred + Vector3.new(0, up, 0)
        end
    end
    return pred
end

local function screenDist(plr)
    local st = TargetState[plr]
    if not st then return nil end
    local s = toScreen(st.root.Position)
    if not s then return nil end
    return (s - workspace.CurrentCamera.ViewportSize / 2).Magnitude
end

-- Friends and players you chose to ignore are skipped when the assist picks a target on its own.
-- (Friend status comes from a web lookup, so it is fetched once in the background and remembered.)
local FriendCache = {}
local function isExcluded(plr)
    if AssistSettings.Ignored[plr.Name] then return true end
    if AssistSettings.SkipFriends then
        local f = FriendCache[plr.UserId]
        if f == nil then
            FriendCache[plr.UserId] = false
            task.spawn(function()
                local ok, res = pcall(function() return p:IsFriendsWith(plr.UserId) end)
                FriendCache[plr.UserId] = (ok and res) and true or false
            end)
            return false
        end
        return f
    end
    return false
end

-- Picks the target and keeps it (sticky) so aim doesn't flicker between players.
local function pickTarget()
    if AssistSettings.TargetPlayer ~= "Auto (Closest)" then
        local plr = Players:FindFirstChild(AssistSettings.TargetPlayer)
        return plr and TargetState[plr] and plr or nil
    end
    local radius = AssistSettings.Effectiveness
    local best, bestD = nil, radius
    for plr in pairs(TargetState) do
        if plr ~= p and not isExcluded(plr) then
            local d = screenDist(plr)
            if d and d < bestD then best, bestD = plr, d end
        end
    end
    if lockedPlr and TargetState[lockedPlr] and not isExcluded(lockedPlr) then
        local ld = screenDist(lockedPlr)
        if ld and ld < radius * 1.15 and (not best or ld <= bestD * 1.6) then
            return lockedPlr
        end
    end
    lockedPlr = best
    return best
end

local currentBestTarget = nil -- legacy modes only
local lastPositions = {}
local MarkerUpdate = function() end -- assigned by the UI block

-- Lets an aimed click pass THROUGH any UI sitting on top of the aim point, so it lands on the
-- world behind it instead of pressing a button or panel. Anything at that spot that would swallow
-- the click (buttons, text boxes, scrolling lists, Active GUI -- including this script's own panels)
-- is made non-interactive just for the moment of the click, then put back exactly as it was.
-- Purely decorative GUI (inactive frames, labels, the glow layer, the aim marker) is never touched.
local GuiService = game:GetService("GuiService")
local passThrough = {} -- obj -> { n = users, interactable = original, active = original }

local function freeClickPath(x, y, subtractInset)
    local freed = {}
    pcall(function()
        local inset = (subtractInset == false) and Vector2.new(0, 0) or GuiService:GetGuiInset()
        local gx, gy = x - inset.X, y - inset.Y
        local function collect(list)
            for _, obj in ipairs(list) do
                if obj.Interactable and (obj:IsA("GuiButton") or obj:IsA("TextBox") or obj:IsA("ScrollingFrame") or obj.Active) then
                    local st = passThrough[obj]
                    if not st then
                        st = { n = 0, interactable = obj.Interactable, active = obj.Active }
                        passThrough[obj] = st
                        pcall(function()
                            obj.Interactable = false
                            obj.Active = false
                        end)
                    end
                    st.n += 1
                    freed[#freed + 1] = obj
                end
            end
        end
        collect(p.PlayerGui:GetGuiObjectsAtPosition(gx, gy))
        pcall(function() collect(game:GetService("CoreGui"):GetGuiObjectsAtPosition(gx, gy)) end)
    end)
    return freed
end

-- put the UI back a few frames AFTER the click's release, so the input has certainly been processed
local function restoreClickPath(freed)
    if not freed or #freed == 0 then return end
    task.delay(0.1, function()
        for _, obj in ipairs(freed) do
            local st = passThrough[obj]
            if st then
                st.n -= 1
                if st.n <= 0 then
                    passThrough[obj] = nil
                    pcall(function()
                        obj.Interactable = st.interactable
                        obj.Active = st.active
                    end)
                end
            end
        end
    end)
end

-- While a macro was started from an on-screen button, unaimed clicks go to the middle of the screen instead of
-- wherever your finger last was (which is on the button itself). nil for keyboard / controller use.
local MacroCursorOverride = nil

-- How the macros "click": "Mouse" = virtual mouse clicks (PC), "Tap" = a virtual finger tap on the middle of the screen
-- (where the shift-lock reticle is), "Tool" = activate the weapon in your hand directly, with no screen input at all.
-- Touch-only devices start on "Tap" so the script never moves a mouse cursor that doesn't exist there.
local ClickMode = "Mouse"
if UIS.TouchEnabled and not UIS.MouseEnabled then ClickMode = "Tap" end
local TapTouchId = 77

-- Called by the macro engine at the exact moment of each aimed click.
local function getAimScreenPos(weapon)
    if AssistSettings.Solver then
        local plr = pickTarget()
        local world = plr and predictFor(plr, weapon)
        local screen = world and toScreen(world)
        if not screen then return nil end
        local l = MacroCursorOverride or UIS:GetMouseLocation()
        return Vector2.new(l.X, l.Y):Lerp(screen, AssistSettings.Smoothing)
    end
    return currentBestTarget
end

RunService.RenderStepped:Connect(function(dt)
    local now = clock()
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= p then updateTarget(plr, now) end
    end

    if not AssistSettings.Enabled then
        currentBestTarget = nil
        MarkerUpdate(nil)
        return
    end

    if AssistSettings.Solver then
        currentBestTarget = nil
        local screen = nil
        if AssistSettings.Marker then
            local plr = pickTarget()
            local world = plr and predictFor(plr, "golden")
            screen = world and toScreen(world)
        end
        MarkerUpdate(screen)
        return
    end

    MarkerUpdate(nil)
    -- legacy (Classic / Aggressive): original behavior
    local cam = workspace.CurrentCamera
    local closest, shortest = nil, AssistSettings.Effectiveness
    local center = cam.ViewportSize / 2
    local safeDt = math.max(dt, 0.001)
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= p and ((AssistSettings.TargetPlayer == "Auto (Closest)" and not isExcluded(plr)) or plr.Name == AssistSettings.TargetPlayer) then
            local char = plr.Character
            local root = char and char:FindFirstChild("HumanoidRootPart")
            local hum = char and char:FindFirstChildOfClass("Humanoid")
            if root and hum and hum.Health > 0 then
                local pos = root.Position
                local last = lastPositions[plr] or pos
                lastPositions[plr] = pos
                local predicted = pos + ((pos - last) / safeDt) * AssistSettings.Prediction
                local sp, onScreen = cam:WorldToViewportPoint(predicted)
                if onScreen then
                    local v = Vector2.new(sp.X, sp.Y)
                    local d = (v - center).Magnitude
                    if d < shortest then shortest, closest = d, v end
                end
            end
        end
    end
    if closest then
        currentBestTarget = currentBestTarget and currentBestTarget:Lerp(closest, AssistSettings.Smoothing) or closest
    else
        currentBestTarget = nil
    end
end)

Players.PlayerRemoving:Connect(function(plr)
    lastPositions[plr] = nil
    TargetState[plr] = nil
    if lockedPlr == plr then lockedPlr = nil end
end)

-- ==========================================
-- TIMESTAMP MACRO ENGINE
-- Every event has an absolute time (ms) from trigger. Each Heartbeat fires
-- all events that are due, so delays never accumulate and several events
-- can land in one frame. A down and its matching up never share a frame,
-- except 0ms taps, which fire together like the hardware macro does.
-- ==========================================
local Engine = { run = nil, conn = nil }

local function sendEvent(e, run)
    pcall(function()
        if e.kind == "key" then
            VIM:SendKeyEvent(e.down, e.code, false, game)
        else
            if e.down then
                local x, y
                if e.aim then registerShot(e.aim) end -- lets the speed learner see every aimed shot
                local aimPos = e.aim and AssistSettings.Enabled and ClickMode ~= "Tool" and getAimScreenPos(e.aim)
                if aimPos then
                    x, y = aimPos.X, aimPos.Y
                    -- only the assist's aimed snowball clicks go through any UI on top of the aim
                    -- point, to the world behind it, instead of pressing the UI
                    run.freed = run.freed or {}
                    run.freed[e.id] = freeClickPath(x, y)
                else
                    -- no cursor on touch devices: the "cursor" is the reticle in the middle of the screen
                    local cam = workspace.CurrentCamera
                    local l = MacroCursorOverride
                        or ((ClickMode ~= "Mouse" and cam) and Vector2.new(cam.ViewportSize.X * 0.5, cam.ViewportSize.Y * 0.5))
                        or UIS:GetMouseLocation()
                    x, y = l.X, l.Y
                    if MacroCursorOverride or ClickMode == "Tap" then -- a click we placed ourselves: let it pass through any UI on top of it
                        run.freed = run.freed or {}
                        run.freed[e.id] = freeClickPath(x, y)
                    end
                end
                run.pos[e.id] = { x, y }
            end
            local pos = run.pos[e.id]
            if e.code == 0 and ClickMode == "Tool" then
                -- no screen input: activate whatever you're holding, the way a tap would
                if e.down then
                    local tool = p.Character and p.Character:FindFirstChildOfClass("Tool")
                    if tool then pcall(function() tool:Activate() end) end
                end
            elseif e.code == 0 and ClickMode == "Tap" then
                local okT = pcall(function() VIM:SendTouchEvent(TapTouchId, e.down and 0 or 2, pos[1], pos[2]) end)
                if not okT then VIM:SendMouseButtonEvent(pos[1], pos[2], e.code, e.down, game, 0) end
            else
                VIM:SendMouseButtonEvent(pos[1], pos[2], e.code, e.down, game, 0)
            end
            if not e.down and run.freed and run.freed[e.id] then
                restoreClickPath(run.freed[e.id])
                run.freed[e.id] = nil
            end
        end
    end)
    run.held[e.id] = e.down and e or nil
end

local function releaseHeld(run)
    for id, e in pairs(run.held) do
        sendEvent({ kind = e.kind, code = e.code, down = false, id = id }, run)
    end
    run.held = {}
end

local function stopEngine()
    if Engine.conn then
        Engine.conn:Disconnect()
        Engine.conn = nil
    end
    local run = Engine.run
    Engine.run = nil
    if run then releaseHeld(run) end
end

local function stepEngine()
    local run = Engine.run
    if not run then return end

    local now = (clock() - run.t0) * 1000
    local downThisFrame = {}
    local events = run.events

    while run.i <= #events do
        local e = events[run.i]
        if e.t > now then break end
        if not e.down and not e.instant and downThisFrame[e.id] then break end -- push up to next frame (0ms taps excepted)
        sendEvent(e, run)
        if e.down then downThisFrame[e.id] = true end
        run.i = run.i + 1
    end

    if run.i > #events then
        stopEngine()
    end
end

local INSTANCE_ID = tostring(os.clock()) .. "-" .. tostring(math.random(1, 1e9))
pcall(function() getgenv().__ExoticaInstance = INSTANCE_ID end)
local function isCurrentInstance() -- false in an older copy of the script that was run again later
    local ok, v = pcall(function() return getgenv().__ExoticaInstance end)
    return (not ok) or v == nil or v == INSTANCE_ID
end

local function startSequence(events)
    if not isCurrentInstance() then return end -- an older copy of the script must not fire macros too
    stopEngine() -- retrigger = clean restart, held inputs released first
    Engine.run = { events = events, i = 1, t0 = clock(), held = {}, pos = {} }
    Engine.conn = RunService.Heartbeat:Connect(stepEngine)
    stepEngine() -- fire first event immediately, zero added latency
end

-- Sequence builder: each step is down at cursor, up at cursor+hold,
-- next step at cursor+hold+gap (times in ms).
local function newSeq()
    local s = { events = {}, t = 0, n = 0 }
    local function add(kind, code, hold, gap, aim)
        s.n = s.n + 1
        table.insert(s.events, { t = s.t, kind = kind, code = code, down = true, id = s.n, aim = aim })
        table.insert(s.events, { t = s.t + hold, kind = kind, code = code, down = false, id = s.n, instant = (hold == 0) })
        s.t = s.t + hold + (gap or 0)
        return s
    end
    function s.key(code, hold, gap) return add("key", code, hold, gap) end
    function s.click(hold, gap, aim) return add("mouse", 0, hold, gap, aim) end
    function s.mouse(btn, hold, gap) return add("mouse", btn, hold, gap) end
    return s
end

local B64_CHARS = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
local B64_LOOKUP = {}
for i = 1, 64 do B64_LOOKUP[B64_CHARS:sub(i, i)] = i - 1 end

local function b64decode(data)
    data = data:gsub("[^%w%+/=]", "")
    local out, n = {}, 0
    for i = 1, #data, 4 do
        local c1, c2, c3, c4 = data:sub(i, i), data:sub(i + 1, i + 1), data:sub(i + 2, i + 2), data:sub(i + 3, i + 3)
        local a, b = B64_LOOKUP[c1], B64_LOOKUP[c2]
        local c, d = B64_LOOKUP[c3], B64_LOOKUP[c4]
        if not (a and b) then break end
        local v = a * 262144 + b * 4096 + (c or 0) * 64 + (d or 0)
        n = n + 1
        if c3 == "=" or not c then
            out[n] = string.char(math.floor(v / 65536))
        elseif c4 == "=" or not d then
            out[n] = string.char(math.floor(v / 65536), math.floor(v / 256) % 256)
        else
            out[n] = string.char(math.floor(v / 65536), math.floor(v / 256) % 256, v % 256)
        end
    end
    return table.concat(out)
end

local soundAssets = {}
local function loadSoundAsset(key)
    if soundAssets[key] ~= nil then return soundAssets[key] or nil end
    local asset = nil
    pcall(function()
        local getAsset = getcustomasset or getsynasset
        local info = SOUND_DATA[key]
        if not (getAsset and writefile and readfile and isfile and info) then return end
        local path = "Exotica_" .. key .. "." .. (info.ext or "mp3")
        local okSize = false
        if isfile(path) then
            local existing = readfile(path)
            okSize = existing ~= nil and #existing == info.size
        end
        if not okSize then
            writefile(path, b64decode(info.data))
        end
        asset = getAsset(path)
    end)
    soundAssets[key] = asset or false
    return asset
end

-- plays one sound; after maxLen seconds it fades out and cleans itself up
local function playSound(id, volume, speed, maxLen)
    if not isCurrentInstance() then return end -- an older copy of the script must not play sounds too
    pcall(function()
        local s = Instance.new("Sound")
        s.SoundId = id
        s.Volume = volume or 0.6
        s.PlaybackSpeed = speed or 1
        s.Parent = SoundService
        s:Play()
        task.delay(maxLen or 10, function()
            pcall(function() TweenService:Create(s, TweenInfo.new(0.12), { Volume = 0 }):Play() end)
            task.delay(0.15, function() s:Destroy() end)
        end)
    end)
end

-- plays one of the embedded sounds. Already-loaded sounds play instantly; the first
-- time, it loads (decode + write the file) and then plays.
local function playEmbedded(key, volume, speed, maxLen, force)
    if not SoundsEnabled and not force then return end
    local asset = soundAssets[key]
    if asset ~= nil then
        local id = asset or (FALLBACK_KEYS[key] and FALLBACK_SOUND)
        if id then playSound(id, volume, speed, maxLen) end
        return
    end
    task.spawn(function()
        local id = loadSoundAsset(key) or (FALLBACK_KEYS[key] and FALLBACK_SOUND)
        if id then playSound(id, volume, speed, maxLen) end
    end)
end

-- Per-element fade (no CanvasGroup, so nothing gets dimmed). Build it AFTER the card's
-- contents exist: it remembers every element's normal transparency and can fade to/from it.
local function makeFader(root)
    local list = {}
    local function add(inst, prop)
        table.insert(list, { inst = inst, prop = prop, orig = inst[prop] })
    end
    local function collect(inst)
        if inst:IsA("GuiObject") then
            add(inst, "BackgroundTransparency")
            if inst:IsA("TextLabel") or inst:IsA("TextButton") or inst:IsA("TextBox") then
                add(inst, "TextTransparency")
            end
            if inst:IsA("ScrollingFrame") then
                add(inst, "ScrollBarImageTransparency")
            end
        elseif inst:IsA("UIStroke") then
            add(inst, "Transparency")
        end
    end
    collect(root)
    for _, d in ipairs(root:GetDescendants()) do collect(d) end

    local fader = {}
    function fader.hideNow()
        for _, it in ipairs(list) do it.inst[it.prop] = 1 end
    end
    function fader.set(visible, time)
        local last
        for _, it in ipairs(list) do
            last = tween(it.inst, { [it.prop] = visible and it.orig or 1 }, time)
        end
        return last
    end
    return fader
end

-- Show/hide controller for a card: fade + gentle scale, safe against rapid toggling.
local function makePanelController(frame, uiScale, fader)
    local ctrl = { open = false }
    local token = 0
    frame.Visible = false
    fader.hideNow()

    function ctrl.show(silent)
        token += 1
        ctrl.open = true
        frame.Visible = true
        uiScale.Scale = 0.92
        fader.set(true, 0.4)
        tween(uiScale, { Scale = 1 }, 0.5, Enum.EasingStyle.Back)
        if not silent then playEmbedded("panel", 0.8, 1, 1) end
    end
    function ctrl.hide(silent)
        if not ctrl.open then return end
        token += 1
        local mine = token
        ctrl.open = false
        if not silent then playEmbedded("panel", 0.8, 0.82, 1) end
        fader.set(false, 0.25)
        tween(uiScale, { Scale = 0.94 }, 0.25)
        task.delay(0.27, function()
            if token == mine then frame.Visible = false end
        end)
    end
    function ctrl.toggle()
        if ctrl.open then ctrl.hide() else ctrl.show() end
    end
    return ctrl
end

local function makeDraggable(handle, target)
    local state = { moved = false }
    local dragging, dragStart, startPos = false, nil, nil

    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging, state.moved = true, false
            dragStart, startPos = input.Position, target.Position
            local conn
            conn = input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                    if conn then conn:Disconnect() end
                end
            end)
        end
    end)

    UIS.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            if delta.Magnitude > 5 then state.moved = true end
            target.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)

    return state
end

    -- A full-width row with a label, a live value readout, and a draggable track.
    -- onChange fires with the new value on every drag update (press, move, release).
    local function makeSlider(labelText, min, max, initial, formatFn, onChange)
        local row = glassRow(58)

        newLabel(row, {
            Size = UDim2.new(0.6, 0, 0, 18),
            Position = UDim2.new(0, 14, 0, 7),
            Font = Enum.Font.GothamMedium,
            Text = labelText,
            TextSize = 13,
            TextXAlignment = Enum.TextXAlignment.Left,
        })

        local valueLabel = newLabel(row, {
            Size = UDim2.new(0.35, 0, 0, 18),
            Position = UDim2.new(0.65, -14, 0, 7),
            Font = Enum.Font.GothamBold,
            Text = formatFn(initial),
            TextSize = 12,
            TextColor3 = COLORS.accent,
            TextXAlignment = Enum.TextXAlignment.Right,
        })

        local track = Instance.new("Frame")
        track.Size = UDim2.new(1, -28, 0, 6)
        track.Position = UDim2.new(0, 14, 0, 39)
        track.BackgroundColor3 = COLORS.white
        track.BackgroundTransparency = 0.78
        track.BorderSizePixel = 0
        track.Parent = row
        corner(track, 3)

        local startFrac = math.clamp((initial - min) / (max - min), 0, 1)

        local fill = Instance.new("Frame")
        fill.Size = UDim2.new(startFrac, 0, 1, 0)
        fill.BackgroundColor3 = COLORS.blue
        fill.BorderSizePixel = 0
        fill.Parent = track
        corner(fill, 3)

        local handle = Instance.new("Frame")
        handle.Size = UDim2.new(0, 20, 0, 20)
        handle.AnchorPoint = Vector2.new(0.5, 0.5)
        handle.Position = UDim2.new(startFrac, 0, 0.5, 0)
        handle.BackgroundColor3 = COLORS.white
        handle.BorderSizePixel = 0
        handle.Parent = track
        corner(handle, 10)
        stroke(handle, Color3.fromRGB(0, 0, 0), 0.85, 1)

        local lastStep = math.floor(startFrac * 20)

        -- Moves the fill/handle/label to match a value without being dragged,
        -- used to sync the slider when a mode preset is applied.
        local function setValue(value)
            local frac = math.clamp((value - min) / (max - min), 0, 1)
            fill.Size = UDim2.new(frac, 0, 1, 0)
            handle.Position = UDim2.new(frac, 0, 0.5, 0)
            valueLabel.Text = formatFn(value)
        end

        local function setFromX(absX)
            local trackPos, trackSize = track.AbsolutePosition.X, track.AbsoluteSize.X
            if trackSize <= 0 then return end
            local frac = math.clamp((absX - trackPos) / trackSize, 0, 1)
            local value = min + frac * (max - min)
            local step = math.floor(frac * 20)
            if step ~= lastStep then
                lastStep = step
                playEmbedded("tick", 0.4, 0.85 + frac * 0.5, 0.5)
            end
            fill.Size = UDim2.new(frac, 0, 1, 0)
            handle.Position = UDim2.new(frac, 0, 0.5, 0)
            valueLabel.Text = formatFn(value)
            onChange(value)
        end

        local dragging = false
        track.InputBegan:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                dragging = true
                container.ScrollingEnabled = false -- so dragging the slider doesn't also scroll the list
                setFromX(input.Position.X)
            end
        end)
        UIS.InputChanged:Connect(function(input)
            if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
                setFromX(input.Position.X)
            end
        end)
        UIS.InputEnded:Connect(function(input)
            if dragging and (input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch) then
                dragging = false
                container.ScrollingEnabled = true
            end
        end)

        return row, setValue
    end

    -- On-screen macro buttons, for players with no keyboard or controller. Add a button for each combo, drag it
    -- where you like ("Move mode"), then lock it so a tap runs the macro.
    ------------------------------------------------------------
    local OSB = { editMode = false, buttons = {}, saved = {}, spec = {
        Snowball = { icon = "❄️", label = "Snowball", color = COLORS.accent, dx = 0.86, dy = 0.30 },
        Board = { icon = "🛹", label = "Board", color = COLORS.accent2, dx = 0.86, dy = 0.46 },
        RPG = { icon = "🚀", label = "RPG", color = COLORS.warn, dx = 0.86, dy = 0.62 },
    } }
    OSB.order = { "Snowball", "Board", "RPG" }
    OSB.file = "ExoticaMacros_buttons.json"
    OSB.save = function()
        local data = {}
        for _, name in ipairs(OSB.order) do
            local b = OSB.buttons[name]
            local prev = OSB.saved[name] or {}
            data[name] = {
                on = b ~= nil,
                x = b and b.Position.X.Scale or prev.x,
                y = b and b.Position.Y.Scale or prev.y,
            }
        end
        OSB.saved = data
        AutoFixMemory.osb = data
        pcall(function()
            if writefile then writefile(OSB.file, game:GetService("HttpService"):JSONEncode(data)) end
        end)
    end
    OSB.load = function()
        local ok, data = pcall(function()
            if isfile and isfile(OSB.file) then return game:GetService("HttpService"):JSONDecode(readfile(OSB.file)) end
            return AutoFixMemory.osb
        end)
        if ok and type(data) == "table" then OSB.saved = data end
    end
    OSB.refresh = function()
        for name, b in pairs(OSB.buttons) do
            local st = b:FindFirstChildOfClass("UIStroke")
            if st then
                st.Color = OSB.editMode and COLORS.warn or OSB.spec[name].color
                st.Transparency = OSB.editMode and 0.05 or 0.3
            end
            b.Text = OSB.spec[name].icon .. "\n" .. (OSB.editMode and "Drag me" or OSB.spec[name].label)
        end
    end
    OSB.fire = function(name, b)
        local sc = b:FindFirstChildOfClass("UIScale")
        if sc then
            sc.Scale = 0.88
            tween(sc, { Scale = 1 }, 0.18, Enum.EasingStyle.Back)
        end
        -- unaimed clicks of this macro go to the middle of the screen, not to the button under your finger
        local cam = workspace.CurrentCamera
        if cam then
            local center = Vector2.new(cam.ViewportSize.X * 0.5, cam.ViewportSize.Y * 0.5)
            MacroCursorOverride = center
            task.delay(5, function() if MacroCursorOverride == center then MacroCursorOverride = nil end end)
        end
        warn("[Exotica] screen button:", name)
        if triggerMacro then
            local ok, err = pcall(triggerMacro, name)
            if not ok then warn("[Exotica] screen button failed:", tostring(err)) end
        end
    end
    OSB.make = function(name)
        if OSB.buttons[name] then return end
        local spec, saved = OSB.spec[name], OSB.saved[name]
        local b = Instance.new("TextButton")
        b.Name = "OSB_" .. name
        b.Size = UDim2.new(0, 68, 0, 68)
        b.AnchorPoint = Vector2.new(0.5, 0.5)
        b.Position = UDim2.new((saved and saved.x) or spec.dx, 0, (saved and saved.y) or spec.dy, 0)
        b.BackgroundColor3 = COLORS.glass
        b.BackgroundTransparency = 0.15
        b.BorderSizePixel = 0
        b.AutoButtonColor = false
        b.Font = Enum.Font.GothamBold
        b.TextSize = 12
        b.TextColor3 = COLORS.white
        b.ZIndex = 60
        b.Parent = sg
        corner(b, 34)
        stroke(b, spec.color, 0.3, 2)
        local sc = Instance.new("UIScale")
        sc.Parent = b
        OSB.buttons[name] = b
        -- A quick tap runs the macro. To move the button: press and hold for a moment, then drag (or turn Move mode on).
        local down, grabbed, moved, pressedAt, startMouse, startPos = false, false, false, 0, Vector2.new(), b.Position
        local myInput = nil -- the exact finger / mouse press that started on this button
        b.InputBegan:Connect(function(i)
            local t = i.UserInputType
            if (t == Enum.UserInputType.MouseButton1 or t == Enum.UserInputType.Touch) and not down then
                myInput = i
                down, grabbed, moved = true, false, false
                pressedAt = os.clock()
                startMouse = Vector2.new(i.Position.X, i.Position.Y)
                startPos = b.Position
                local conn
                conn = i.Changed:Connect(function()
                    local st = i.UserInputState
                    if st == Enum.UserInputState.End or st == Enum.UserInputState.Cancel then
                        conn:Disconnect()
                        down = false
                        sc.Scale = 1
                        if grabbed then
                            OSB.save()
                            OSB.refresh()
                        elseif st == Enum.UserInputState.End and not moved and not OSB.editMode and os.clock() - pressedAt < 1.2 then
                            OSB.fire(name, b)
                        end
                    end
                end)
            end
        end)
        local moveConn
        moveConn = game:GetService("UserInputService").InputChanged:Connect(function(i)
            -- only the finger that pressed this button counts: a second finger on the movement stick must not cancel the tap
            local mine = down and myInput and (i == myInput
                or (myInput.UserInputType == Enum.UserInputType.MouseButton1 and i.UserInputType == Enum.UserInputType.MouseMovement))
            if mine then
                local d = Vector2.new(i.Position.X, i.Position.Y) - startMouse
                if d.Magnitude > 12 then
                    if OSB.editMode or (os.clock() - pressedAt) >= 0.3 then
                        if not grabbed then
                            grabbed = true
                            sc.Scale = 1.12
                            local st = b:FindFirstChildOfClass("UIStroke")
                            if st then st.Color = COLORS.warn st.Transparency = 0.05 end
                        end
                    else
                        moved = true -- swiped before the hold: not a tap, not a move
                    end

        -- find the real hotbar (default Roblox backpack, or a custom one in PlayerGui)

        -- Finds the hotbar by its SHAPE, not by number labels (phones don't show them): a row of
        -- equal-size, roughly square cells on screen. Counting left to right = slot 1..9, then 0.
        -- If number labels happen to be printed on the cells (PC), they are used to confirm the
        -- numbering, but nothing depends on them.
        local rowPitch, rowRef, lastRowInfo = 64, nil, nil
        local invWatchDefault, invWatchGrid, autoOpen = nil, nil, false -- backpack-open detection
        local hotCellsByPos = {} -- hotbar position (1..10) -> its cell, as last resolved

        local function digitPos(s) return (s == "0") and 10 or tonumber(s) end
        local function nn(s) return (tostring(s):lower():gsub("[^%w]", "")) end

        local function chainShown(o)
            local cur = o
            while cur and cur ~= game do
                if cur:IsA("GuiObject") and not cur.Visible then return false end
                if cur:IsA("ScreenGui") and not cur.Enabled then return false end
                cur = cur.Parent
            end
            return true
        end

        local function resolveSlotFrames()
            local screen = sg.AbsoluteSize
            local roots = {}
            local pg = p:FindFirstChildOfClass("PlayerGui")
            if pg then roots[#roots + 1] = pg end
            pcall(function() roots[#roots + 1] = game:GetService("CoreGui") end)

            -- names of the tools you hold, used only to tell the real hotbar from other square rows
            local toolNames = {}
            local bp = p:FindFirstChildOfClass("Backpack")
            if bp then for _, t in ipairs(bp:GetChildren()) do if t:IsA("Tool") then toolNames[nn(t.Name)] = true end end end
            if p.Character then
                for _, t in ipairs(p.Character:GetChildren()) do if t:IsA("Tool") then toolNames[nn(t.Name)] = true end end
            end

            -- every visible, roughly square, slot-sized element, grouped by its parent
            local groups, order = {}, {}
            for _, root in ipairs(roots) do
                for _, d in ipairs(root:GetDescendants()) do
                    if d:IsA("GuiObject") and d.Visible then
                        local s = d.AbsoluteSize
                        if s.X >= 24 and s.X <= 260 and s.Y >= 24 and s.Y <= 260
                            and s.X / s.Y >= 0.6 and s.X / s.Y <= 1.6 and d.Parent then
                            local list = groups[d.Parent]
                            if not list then list = {} groups[d.Parent] = list order[#order + 1] = d.Parent end
                            list[#list + 1] = d
                        end
                    end
                end
            end

            local bestCells, bestScore, bestParent, bestEntry = nil, -1e9, nil, nil
            local others = {}
            for _, par in ipairs(order) do
                local list = groups[par]
                if #list >= 3 and not par:IsDescendantOf(sg) and chainShown(par) then
                    -- biggest set of cells sharing the same row (Y) and about the same size
                    local cluster = nil
                    for _, c in ipairs(list) do
                        local cur = {}
                        for _, o in ipairs(list) do
                            if math.abs(o.AbsolutePosition.Y - c.AbsolutePosition.Y) <= 28
                                and math.abs(o.AbsoluteSize.X - c.AbsoluteSize.X) <= c.AbsoluteSize.X * 0.35 then
                                cur[#cur + 1] = o
                            end
                        end
                        if not cluster or #cur > #cluster then cluster = cur end
                    end
                    table.sort(cluster, function(x, y) return x.AbsolutePosition.X < y.AbsolutePosition.X end)
                    local cells = {}
                    for _, o in ipairs(cluster) do -- drop overlapping duplicates
                        local last = cells[#cells]
                        if not last or o.AbsolutePosition.X - last.AbsolutePosition.X > last.AbsoluteSize.X * 0.5 then
                            cells[#cells + 1] = o
                        end
                    end
                    if #cells >= 3 then
                        local hits = 0
                        for _, cell in ipairs(cells) do
                            for _, d in ipairs(cell:GetDescendants()) do
                                if (d:IsA("TextLabel") or d:IsA("TextButton")) and toolNames[nn(d.Text)] then
                                    hits = hits + 1
                                    break
                                end
                            end
                        end
                        local rowY = cells[1].AbsolutePosition.Y
                        local pn = par.Name:lower()
                        local gn = par.Parent and par.Parent.Name:lower() or ""
                        -- The hotbar sits at the BOTTOM of the screen, so lower on screen wins by a wide
                        -- margin. This keeps it from locking onto the backpack's item grid higher up.
                        local score = #cells * 5 + hits * 10 + (rowY / math.max(screen.Y, 1)) * 600
                        if pn:find("hotbar", 1, true) or gn:find("hotbar", 1, true) then score = score + 300 end
                        if par:IsA("ScrollingFrame") or pn:find("inventory", 1, true) or gn:find("inventory", 1, true) then
                            score = score - 200
                        end
                        local entry = { par = par, cells = cells, hits = hits, y = rowY, h = cells[1].AbsoluteSize.Y }
                        others[#others + 1] = entry
                        -- only something that looks like a real hotbar may be chosen as the hotbar: named for it,
                        -- or 6+ cells. (Otherwise small button rows such as the phone's home
                        -- screen get picked whenever the real hotbar is briefly hidden.)
                        local plausible = pn:find("hotbar", 1, true) or gn:find("hotbar", 1, true) or #cells >= 6
                        if plausible and score > bestScore then bestCells, bestScore, bestParent, bestEntry = cells, score, par, entry end
                    end
                end
            end

            -- Backpack-open detection (so the glow can appear on its own): Roblox's default Inventory
            -- frame, or a grid of item cells sitting above the hotbar. Remembered once seen, so
            -- opening it again is noticed quickly even after it's hidden.
            pcall(function()
                local inv = game:GetService("CoreGui").RobloxGui.Backpack.Inventory
                if inv:IsA("GuiObject") then invWatchDefault = inv end
            end)
            if bestEntry then
                local grid = nil
                for _, e in ipairs(others) do
                    if e.par ~= bestParent and e.hits >= 1 and e.y < bestEntry.y - bestEntry.h * 0.8 then
                        if not grid or #e.cells > #grid.cells then grid = e end
                    end
                end
                if grid then invWatchGrid = grid.par end
            end

            if not bestCells then
                rowRef = nil
                for _, g in ipairs(glows) do g.frame = nil end
                if lastRowInfo ~= "none" then
                    lastRowInfo = "none"
                    print("[Exotica] slot row: no hotbar found on screen yet")
                end
                return
            end

            -- numbering: each cell's OWN number label says which slot it is, so a cell that is missing or drawn
            -- a little differently (for example the one you're holding) can't shift every slot after it
            local byPos, labeled = {}, {}
            for _, cell in ipairs(bestCells) do
                for _, d in ipairs(cell:GetDescendants()) do
                    if (d:IsA("TextLabel") or d:IsA("TextButton")) and #d.Text == 1 and d.Text:match("%d") then
                        local pos = digitPos(d.Text)
                        if not byPos[pos] then
                            byPos[pos] = cell
                            labeled[#labeled + 1] = { cell = cell, pos = pos }
                        end
                        break
                    end
                end
            end
            if #labeled < 2 then
                -- no usable labels: number them left to right
                byPos, labeled = {}, {}
                for i, cell in ipairs(bestCells) do byPos[i] = cell end
                labeled[1] = { cell = bestCells[1], pos = 1 }
            end

            -- spacing between slots: from the labelled cells when we have them, else the median gap
            local pitch
            if #labeled >= 2 then
                table.sort(labeled, function(x, y) return x.pos < y.pos end)
                local gaps = {}
                for i = 2, #labeled do
                    local dp = labeled[i].pos - labeled[i - 1].pos
                    if dp > 0 then
                        gaps[#gaps + 1] = (labeled[i].cell.AbsolutePosition.X - labeled[i - 1].cell.AbsolutePosition.X) / dp
                    end
                end
                table.sort(gaps)
                pitch = gaps[1] and gaps[math.ceil(#gaps / 2)] or nil
            end
            if not pitch or pitch < 10 then
                if #bestCells >= 2 then
                    local gaps = {}
                    for i = 2, #bestCells do
                        gaps[#gaps + 1] = bestCells[i].AbsolutePosition.X - bestCells[i - 1].AbsolutePosition.X
                    end
                    table.sort(gaps)
                    pitch = math.max(gaps[math.ceil(#gaps / 2)], 10)
                else
                    pitch = bestCells[1].AbsoluteSize.X + 4
                end
            end
            -- a cell without a label: work out its slot from where it sits next to a labelled one
            local ref0 = labeled[1]
            for _, cell in ipairs(bestCells) do
                local known = false
                for _, c2 in pairs(byPos) do if c2 == cell then known = true break end end
                if not known and ref0 then
                    local pos = ref0.pos + math.floor((cell.AbsolutePosition.X - ref0.cell.AbsolutePosition.X) / pitch + 0.5)
                    if pos >= 1 and pos <= 10 and not byPos[pos] then byPos[pos] = cell end
                end
            end

            for _, g in ipairs(glows) do g.frame = byPos[g.pos] end
            hotCellsByPos = byPos
            rowPitch = pitch
            rowRef = { cell = ref0.cell, pos = ref0.pos }
            local offset = ref0.pos - 1

            local info = bestParent:GetFullName() .. " | cells " .. #bestCells .. " | first slot " .. (1 + offset)
            if info ~= lastRowInfo then
                lastRowInfo = info
                print("[Exotica] slot row found:", info)
            end
        end

        -- the slot's own cell (nil if this slot isn't drawn on screen)
        local function slotFrameFor(g)
            local f = g.frame
            if f and f.Parent then return f end
            return nil
        end

        -- screen rectangle (relative to our ScreenGui) for a slot, or nil to hide its glow
        local function slotRect(g, origin, screen)
            local f = g.frame
            if f and f.Parent and f.AbsoluteSize.X > 4 and f.AbsoluteSize.Y > 4 then
                return f.AbsolutePosition - origin, f.AbsoluteSize
            end
            -- slot not drawn (e.g. an empty slot that's hidden): continue the row with the same spacing
            if rowRef and rowRef.cell.Parent then
                local ref = rowRef.cell
                local x = ref.AbsolutePosition.X + (g.pos - rowRef.pos) * rowPitch
                if x > -rowPitch and x < screen.X then
                    return Vector2.new(x, ref.AbsolutePosition.Y) - origin, ref.AbsoluteSize
                end
            end
            return nil -- hotbar not found: show nothing rather than a wrong spot
        end

        local function readSlotName(g)
            local f = slotFrameFor(g)
            if not f then return nil end
            local label = f:FindFirstChild("ToolName", true)
            if label and (label:IsA("TextLabel") or label:IsA("TextButton")) then
                return label.Text or ""
            end
            for _, d in ipairs(f:GetDescendants()) do -- custom hotbars: any text that isn't the slot number
                if (d:IsA("TextLabel") or d:IsA("TextButton")) and d.Text ~= "" and d.Text ~= g.slot then
                    return d.Text
                end
            end
            return nil
        end

        ------------------------------------------------------------
        -- Inventory scan: find the real tool names in the Backpack / equipped tool and
        -- match them (fuzzily) against the item names in SlotInfo, so exact in-game
        -- names don't have to be guessed.
        ------------------------------------------------------------
        local function norm(s) return (tostring(s):lower():gsub("[^%w]", "")) end
        -- numeric asset id inside an image string, so "rbxassetid://123" == "...asset/?id=123"
        local function imageId(s)
            s = tostring(s or "")
            if s == "" then return nil end
            local best = nil
            for d in s:gmatch("%d+") do
                if not best or #d >= #best then best = d end
            end
            return (best and #best >= 5) and best or nil
        end
        local function tokens(s)
            local t = {}
            for w in tostring(s):lower():gmatch("%a+") do t[#t + 1] = w end
            return t
        end

        local CANDS = {}
        for _, group in ipairs(SlotInfo) do
            for _, entry in ipairs(group.rows) do
                local list = {}
                local cleaned = entry.item:gsub("%b()", "")
                cleaned = cleaned:gsub("%s+[Oo][Rr]%s+", "/") -- "Yin & Yang or Pink Stop Sign" -> two names
                for part in cleaned:gmatch("[^/]+") do
                    part = part:gsub("^%s+", ""):gsub("%s+$", "")
                    if part ~= "" then list[#list + 1] = part end
                end
                CANDS[entry.slot] = list
            end
        end

        -- 0..1 similarity between an in-game tool name and a script item name
        local function score(toolName, cand)
            local nt, nc = norm(toolName), norm(cand)
            if nt == "" or nc == "" then return 0 end
            local tt, ct, tset, cset = tokens(toolName), tokens(cand), {}, {}
            for _, w in ipairs(tt) do tset[w] = true end
            for _, w in ipairs(ct) do cset[w] = true end
            -- Golden vs Normal launcher must never be confused with each other
            local tGold = (tset.golden or tset.gold) and true or false
            local cGold = (cset.golden or cset.gold) and true or false
            if tGold ~= cGold then return 0 end
            -- same for Spiked vs normal ("Spiked Kitty Stanley" must never count as "Kitty Stanley", or the other way round)
            if (nt:find("spik", 1, true) ~= nil) ~= (nc:find("spik", 1, true) ~= nil) then return 0 end
            if nt:find(nc, 1, true) or nc:find(nt, 1, true) then return 1 end
            local overlap = 0
            for w in pairs(cset) do if tset[w] then overlap = overlap + 1 end end
            return overlap / math.max(#ct, #tt, 1)
        end

        local function classify(name)
            local n = nz(name)
            for _, e in ipairs(WEAPONS) do
                local hit
                if e.exact then
                    hit = false
                    for _, k in ipairs(e.exact) do if n == k then hit = true end end
                elseif e.all then
                    hit = true
                    for _, k in ipairs(e.all) do if not n:find(k, 1, true) then hit = false end end
                else
                    hit = hasAny(n, e.any)
                    if hit and e.none and hasAny(n, e.none) then hit = false end
                end
                if hit then return e end
            end
            return nil
        end

        -- a tool's own name can be a skin name ("Purrse of Doom"), so when the name isn't recognised the text the game
        -- stores on the tool (attributes, string values) is tried too, in case it names the base weapon
        local toolClassCache = setmetatable({}, { __mode = "k" })
        local unknownSeen = {}
        WEAPONS.tool = function(t)
            local c = toolClassCache[t]
            if c then return c end
            local e = classify(t.Name)
            if not e then
                pcall(function()
                    for _, v in pairs(t:GetAttributes()) do
                        if type(v) == "string" and not e then e = classify(v) end
                    end
                    for _, ch in ipairs(t:GetChildren()) do
                        if ch:IsA("StringValue") and not e then e = classify(ch.Value) end
                    end
                end)
            end
            if e then toolClassCache[t] = e
            elseif not unknownSeen[t.Name] then
                unknownSeen[t.Name] = true
                local parts = {}
                pcall(function() for k, v in pairs(t:GetAttributes()) do parts[#parts + 1] = tostring(k) .. "=" .. tostring(v) end end)
                warn("[AutoFix] unrecognised tool:", t.Name, #parts > 0 and ("| attributes: " .. table.concat(parts, ", ")) or "")
            end
            return e
        end

        local PLAN = { slotTool = {}, accessory = {}, equip = {}, equipAll = {}, owned = {}, allTools = {} }
        local function planWeapons(tools)
            local rpg, taser, fist
            local dps, acc, equips = {}, {}, {}
            local hasKnuckle = false
            local board
            for _, t in ipairs(tools) do
                local e = WEAPONS.tool(t)
                if e then
                    if e.kind == "loveboard" then
                        board = board or t
                    elseif e.kind == "rpg" then
                        if not rpg or e.tier > rpg.e.tier then rpg = { t = t, e = e } end
                    elseif e.kind == "taser" then taser = taser or t
                    elseif e.kind == "fist" then fist = fist or t
                    elseif e.kind == "dps" then dps[#dps + 1] = { t = t, e = e }
                    elseif e.kind == "acc" then acc[#acc + 1] = { t = t, e = e }
                    elseif e.kind == "equip" then
                        equips[#equips + 1] = { t = t, e = e }
                        if e.group == "knuckle" then hasKnuckle = true end
                    end
                end
            end
            local out = { slotTool = {}, accessory = {}, equip = {}, equipAll = {}, owned = {}, allTools = {} }
            for _, t in ipairs(tools) do out.allTools[#out.allTools + 1] = t end
            for _, t in ipairs(tools) do
                local e0 = WEAPONS.tool(t)
                if e0 and e0.kind ~= "skip" then out.owned[#out.owned + 1] = t end
            end
            for _, q in ipairs(equips) do out.equipAll[#out.equipAll + 1] = q.t end
            if rpg then out.slotTool["5"] = rpg.t end
            if board then out.slotTool["6"] = board end
            if taser then out.slotTool["1"] = taser end

            local pool = {}
            for _, w in ipairs(dps) do
                if not (w.e.noKnuckles and hasKnuckle) then pool[#pool + 1] = w end -- claws never go with knuckles
            end
            table.sort(pool, function(a, b)
                if a.e.tier ~= b.e.tier then return a.e.tier > b.e.tier end
                return a.t.Name < b.t.Name
            end)
            local free = { ["4"] = true, ["0"] = true, ["8"] = not hasKnuckle, ["1"] = (taser == nil) }
            local placed = {}
            for _, w in ipairs(pool) do
                for _, sl in ipairs(w.e.slots) do
                    local k = tostring(sl)
                    if k ~= "1" and free[k] then
                        out.slotTool[k] = w.t
                        free[k] = false
                        placed[w] = true
                        break
                    end
                end
            end
            -- no Taser: the best weapon left over (one that is allowed in more than one slot) takes slot 1
            if free["1"] then
                for _, w in ipairs(pool) do
                    if not placed[w] and #w.e.slots >= 2 then
                        out.slotTool["1"] = w.t
                        out.slot1Fallback = w.t
                        placed[w] = true
                        break
                    end
                end
            end
            -- the Fist uses your knuckles, so with any knuckles it lives in slot 8; without them it's a last resort
            if fist and not out.slotTool["8"] then out.slotTool["8"] = fist end

            table.sort(acc, function(a, b)
                if a.e.tier ~= b.e.tier then return a.e.tier > b.e.tier end
                return a.t.Name < b.t.Name
            end)
            for i = 1, math.min(4, #acc) do out.accessory[i] = acc[i].t end

            local bestOf = {}
            for _, q in ipairs(equips) do
                local cur = bestOf[q.e.group]
                if not cur or q.e.tier > cur.e.tier then bestOf[q.e.group] = q end
            end
            for _, grp in ipairs({ "knuckle", "boots", "cloak", "crown", "gorilla" }) do
                if bestOf[grp] then out.equip[#out.equip + 1] = bestOf[grp].t end
            end
            return out
        end

        local function shortToolName(t)
            local e = WEAPONS.tool(t)
            if e and e.kind == "fist" then return "Fist" end
            return t.Name:sub(1, 14)
        end

        local function scanInventory()
            local tools = {}
            local bp = p:FindFirstChildOfClass("Backpack")
            if bp then
                for _, t in ipairs(bp:GetChildren()) do
                    if t:IsA("Tool") then tools[#tools + 1] = t end
                end
            end
            local ch = p.Character
            if ch then -- the tool you're holding lives in your Character, not the Backpack
                for _, t in ipairs(ch:GetChildren()) do
                    if t:IsA("Tool") then tools[#tools + 1] = t end
                end
            end
            local parts = {}
            for _, t in ipairs(tools) do
                parts[#parts + 1] = t.Name .. "(" .. tostring(imageId(t.TextureId) or "no icon") .. ")"
            end
            toolsDesc = table.concat(parts, ", ")
            local byIcon = {}
            for _, t in ipairs(tools) do
                local tid = imageId(t.TextureId)
                if tid then byIcon[tid] = t.Name end
            end
            toolByIcon = byIcon
            -- decide the best weapon for each variable slot from what you own, and aim the checks at it
            local okPlan, plan = pcall(planWeapons, tools)
            if okPlan and plan then
                PLAN = plan
                for _, slot in ipairs({ "1", "4", "5", "8", "0", "2", "3" }) do
                    PLANNED[slot] = true
                    local t = plan.slotTool[slot]
                    if slot == "2" or slot == "3" then
                        local names = {}
                        for _, a in ipairs(plan.accessory) do names[#names + 1] = a.Name end
                        CANDS[slot] = names
                    else
                        CANDS[slot] = t and { t.Name } or {}
                    end
                    for _, g in ipairs(glows) do
                        if g.slot == slot then
                            g.label = slot .. " · " .. (t and shortToolName(t) or ((SHORT[slot]) or "none"))
                        end
                    end
                end
                if plan.slotTool["6"] then -- Loveboard (or one of its skins) is in slot 6, whatever it's called
                    PLANNED["6"] = true
                    CANDS["6"] = { plan.slotTool["6"].Name }
                end
            end
            local result = {}
            for slot, cands in pairs(CANDS) do
                local set, found, images = {}, false, {}
                for _, t in ipairs(tools) do
                    for _, cand in ipairs(cands) do
                        if score(t.Name, cand) >= 0.6 then
                            set[norm(t.Name)] = true
                            local tid = imageId(t.TextureId)
                            if tid then images[tid] = true end
                            if t.ToolTip and t.ToolTip ~= "" then set[norm(t.ToolTip)] = true end
                            found = true
                            break
                        end
                    end
                end
                if not PLANNED[slot] then
                    for k in pairs(globalSet[slot] or {}) do set[k] = true end
                    for k in pairs(globalImg[slot] or {}) do images[k] = true end
                    for k in pairs(guiSet[slot] or {}) do set[k] = true end -- known name; does NOT mean you own it
                    for k in pairs(guiImg[slot] or {}) do images[k] = true end
                end
                result[slot] = { set = set, found = found, images = images }
            end
            expected = result
        end

        -- last-resort keyword check, used only when the inventory scan found no match for a slot
        local function keywordOk(slot, lower)
            local rule = MATCH[slot]
            if not rule or PLANNED[slot] then return false end
            local hit = false
            for _, w in ipairs(rule.any) do
                if lower:find(w, 1, true) then hit = true break end
            end
            if hit and rule.none then
                for _, w in ipairs(rule.none) do
                    if lower:find(w, 1, true) then hit = false break end
                end
            end
            return hit
        end

        -- icons drawn inside a slot. Roblox's hotbar shows a tool's picture INSTEAD of its name when
        -- the tool has one, so name text alone can't identify those tools (the fist, for example).
        local function readSlotImages(g)
            local f = slotFrameFor(g)
            if not f then return nil end
            local ids = {}
            local function take(d)
                if (d:IsA("ImageLabel") or d:IsA("ImageButton")) and d.Visible then
                    local id = imageId(d.Image)
                    if id then ids[id] = true end
                end
            end
            take(f)
            for _, d in ipairs(f:GetDescendants()) do take(d) end
            return ids
        end

        -- does a displayed name fit one of this slot's script items? (covers display names that
        -- differ a little from the tool's internal name)
        local function nameFitsSlot(slot, name)
            for _, cand in ipairs(CANDS[slot] or {}) do
                if score(name, cand) >= 0.6 then return true end
            end
            return false
        end

        -- Other names these weapons are known by (matched as whole words / parts of words).
        -- A slot holds only a handful of possible weapons, so a short list per slot is enough.
        local SYNONYMS = {
            ["8"] = { "fists", "punch", "punching", "hands", "hand", "knuckles", "brass", "boxing", "glove", "unarmed", "combat", "melee", "brawl" },
            ["4"] = { "yinyang", "stopsign", "pinksign" },
            ["0"] = { "umbrella", "slasher", "claws" },
        }

        -- how many letters differ between two strings (small = a typo / spelling variant)
        local function editDistance(a, b)
            local la, lb = #a, #b
            if math.abs(la - lb) > 2 then return 99 end
            local prev = {}
            for j = 0, lb do prev[j] = j end
            for i = 1, la do
                local cur = { [0] = i }
                for j = 1, lb do
                    local cost = (a:byte(i) == b:byte(j)) and 0 or 1
                    cur[j] = math.min(prev[j] + 1, cur[j - 1] + 1, prev[j - 1] + cost)
                end
                prev = cur
            end
            return prev[lb]
        end

        -- is this text a close enough match for one of the slot's weapons?
        local function closeToSlot(slot, str)
            local n = norm(str)
            if #n < 3 or n:match("^%d+$") then return false end
            local nGold = n:find("gold", 1, true) ~= nil
            for _, cand in ipairs(CANDS[slot] or {}) do
                if score(str, cand) >= 0.6 then return true end
                local nc = norm(cand)
                if #nc >= 4 and (nc:find("gold", 1, true) ~= nil) == nGold
                    and editDistance(n, nc) <= math.max(1, math.floor(#nc / 4)) then
                    return true
                end
            end
            if not PLANNED[slot] then
                for _, syn in ipairs(SYNONYMS[slot] or {}) do
                    if n == syn or (#syn >= 5 and n:find(syn, 1, true)) then return true end
                end
            end
            return false
        end

        -- everything a slot's cell exposes that could be a weapon name: its text (even hidden text),
        -- its own name, attributes, and any value objects hung on it
        local function harvestSlotStrings(g)
            local f = slotFrameFor(g)
            if not f then return nil end
            local list, seen = {}, {}
            local function add(s)
                if type(s) ~= "string" then return end
                local n = norm(s)
                if #n >= 3 and not n:match("^%d+$") and not seen[n] then
                    seen[n] = true
                    list[#list + 1] = s
                end
            end
            local function look(o)
                if o:IsA("TextLabel") or o:IsA("TextButton") then add(o.Text) end
                if o:IsA("StringValue") then add(o.Value) end
                if o:IsA("ObjectValue") and o.Value then add(o.Value.Name) end
                for _, v in pairs(o:GetAttributes()) do add(v) end
            end
            add(f.Name)
            look(f)
            local cnt = 0
            for _, d in ipairs(f:GetDescendants()) do
                cnt = cnt + 1
                if cnt > 60 then break end
                look(d)
            end
            return list
        end

        -- statuses: ok = right weapon in the slot, bad = wrong/empty slot,
        --           missing = weapon isn't in your inventory at all, unknown = couldn't read the slot
        local function checkSlot(g)
            local exp = expected[g.slot]
            local name = readSlotName(g)
            local imgs = readSlotImages(g)
            local n = name and norm(name) or ""

            local okByName = n ~= "" and ((exp and exp.set[n]) or nameFitsSlot(g.slot, name) or keywordOk(g.slot, name:lower()))
            -- also try everything else the slot exposes, plus the name of whichever tool owns the slot's picture
            local strs = harvestSlotStrings(g) or {}
            for id in pairs(imgs or {}) do
                local tn = toolByIcon[id]
                if tn then strs[#strs + 1] = tn end
            end
            if not okByName then
                for _, s in ipairs(strs) do
                    if closeToSlot(g.slot, s) then okByName = true break end
                end
            end
            local okByImage = false
            if exp and exp.images and imgs then
                for id in pairs(imgs) do
                    if exp.images[id] then okByImage = true break end
                end
            end

            local status
            if okByName or okByImage then
                status = "ok"
            elseif exp and exp.found then
                status = (name ~= nil or imgs ~= nil) and "bad" or "unknown"
            elseif name ~= nil or imgs ~= nil then
                status = "missing"
            else
                status = exp and "missing" or "unknown"
            end

            -- accessory slots 2 and 3: whatever you put there is right; only an EMPTY one is a problem
            if g.slot == "2" or g.slot == "3" then
                local filled = (name ~= nil and name ~= "") or (imgs ~= nil and next(imgs) ~= nil)
                if filled then
                    -- only accessory weapons belong here; anything else, even something unknown, is wrong
                    status = (accessoryForeign and accessoryForeign(g)) and "bad" or "ok"
                end
            end

            -- one console line whenever a slot looks wrong, so a mismatch can be diagnosed
            if (status == "bad" or status == "missing") and g.slot ~= "2" and g.slot ~= "3" then
                local have, want = {}, {}
                for id in pairs(imgs or {}) do have[#have + 1] = id end
                for id in pairs(exp and exp.images or {}) do want[#want + 1] = id end
                local msg = string.format("slot %s shows text=%q icons=[%s] | expected icons=[%s] | your tools: %s",
                    g.slot, tostring(name), table.concat(have, ","), table.concat(want, ","), toolsDesc)
                if msg ~= g.lastDbg then
                    g.lastDbg = msg
                    print("[Exotica]", msg)
                end
            end

            if status ~= g.status then
                g.status = status
                local c = (status == "ok") and OK_COLOR
                    or (status == "bad") and BAD_COLOR
                    or (status == "missing") and COLORS.warn
                    or g.base
                g.edge.Color, g.haloEdge.Color = c, c
                g.box.BackgroundColor3, g.tag.TextColor3 = c, c
                local mark = (status == "ok") and "✅ "
                    or (status == "bad") and "❌ "
                    or (status == "missing") and "⚠ "
                    or ""
                g.tag.Text = mark .. g.label
            end
        end

        ------------------------------------------------------------
        -- Auto Slot Fix: when the backpack is open and a needed weapon is sitting in the wrong place
        -- (in the inventory grid, or in another hotbar slot), drag it into the slot it belongs in.
        -- Safe by design: only runs while the backpack is open, only after you've been still for a moment,
        -- never during a macro, tries each slot at most twice per time you open the backpack, and
        -- only moves weapons the scan has actually found in your inventory.
        ------------------------------------------------------------
        local MAXTRIES = 6 -- one try per way of moving an item (see moveWith)
        local autoFix = { busy = false, tries = {}, openSince = 0, nextTick = 0, nextClosed = 0, cool = 0,
            closedCycles = 0, badStreak = 0, forceUntil = 0, lastPrompt = -1000, silent = false, key = nil, method = AutoFixMemory.method, addInset = AutoFixMemory.addInset, calibrated = (AutoFixMemory.addInset ~= nil),
            equipDone = {}, equipTries = {}, rearmAt = 0, relaxUntil = 0,
            lastFailSig = "", lastFailAt = 0 }
        local GuiService_ = game:GetService("GuiService")

        -- Does AbsolutePosition leave the top-bar inset out? Measured once with a throw-away GUI, so
        -- the virtual mouse lands on the exact spot instead of ~36px off.
        local insetNeeded
        local function probeInset()
            if insetNeeded ~= nil then return insetNeeded end
            local inset = GuiService_:GetGuiInset()
            if inset.Y <= 0.5 then insetNeeded = false return false end
            local ok, res = pcall(function()
                local t = Instance.new("ScreenGui")
                t.IgnoreGuiInset = false
                t.ResetOnSpawn = false
                t.Parent = core
                local f = Instance.new("Frame")
                f.Size = UDim2.fromOffset(4, 4)
                f.BackgroundTransparency = 1
                f.Parent = t
                RunService.RenderStepped:Wait()
                RunService.RenderStepped:Wait()
                local y = f.AbsolutePosition.Y
                t:Destroy()
                return y < inset.Y * 0.5
            end)
            insetNeeded = ok and res or false
            return insetNeeded
        end

        local function afLog(...) warn("[AutoFix]", ...) end

        local function toMouse(absCenter, owner)
            local add = autoFix.addInset -- measured by calibrateOffset below, when it has run
            if add == nil then
                local gui = owner and owner:FindFirstAncestorWhichIsA("ScreenGui")
                add = not (gui and gui.IgnoreGuiInset) and probeInset()
            end
            if add then absCenter = absCenter + GuiService_:GetGuiInset() end
            return absCenter
        end

        -- Listens on a cell (and what's inside it) so we can SEE whether the virtual mouse/finger really
        -- reached it: did the pointer enter it, did a press land on it, did a release land on it.
        local function watchHits(obj)
            local rec = { began = 0, ended = 0, entered = false, conns = {} }
            local function hook(o)
                if not o:IsA("GuiObject") then return end
                rec.conns[#rec.conns + 1] = o.InputBegan:Connect(function(i)
                    local t = i.UserInputType
                    if t == Enum.UserInputType.MouseButton1 or t == Enum.UserInputType.Touch then rec.began = rec.began + 1 end
                end)
                rec.conns[#rec.conns + 1] = o.InputEnded:Connect(function(i)
                    local t = i.UserInputType
                    if t == Enum.UserInputType.MouseButton1 or t == Enum.UserInputType.Touch then rec.ended = rec.ended + 1 end
                end)
                rec.conns[#rec.conns + 1] = o.MouseEnter:Connect(function() rec.entered = true end)
            end
            hook(obj)
            local n = 0
            for _, d in ipairs(obj:GetDescendants()) do
                n = n + 1
                if n > 40 then break end
                hook(d)
            end
            function rec.stop() for _, c in ipairs(rec.conns) do c:Disconnect() end end
            return rec
        end

        -- Work out whether the virtual mouse needs the top-bar offset added: hover the item both ways and
        -- keep whichever one the item actually notices. Done once, then remembered.
        local function calibrateOffset(cell)
            if autoFix.addInset ~= nil or autoFix.calibrated then return end
            autoFix.calibrated = true
            local inset = GuiService_:GetGuiInset()
            local c = cell.AbsolutePosition + cell.AbsoluteSize / 2
            SynthDragging = true
            for _, add in ipairs({ true, false }) do
                local rec = watchHits(cell)
                local pt = add and (c + inset) or c
                pcall(function()
                    VIM:SendMouseMoveEvent(c.X + 160, c.Y + 160, game) -- start well outside, so entering is a real event
                    task.wait(0.12)
                    VIM:SendMouseMoveEvent(pt.X, pt.Y, game)
                    task.wait(0.25)
                end)
                local hit = rec.entered
                rec.stop()
                afLog("calibrate: top-bar offset", add and "added" or "not added", "->", hit and "item noticed the pointer" or "no reaction")
                if hit then
                    autoFix.addInset = add
                    AutoFixMemory.addInset = add
                    break
                end
            end
            SynthDragging = false
        end

        local function cellCenter(cell)
            return toMouse(cell.AbsolutePosition + cell.AbsoluteSize / 2, cell)
        end

        -- centre of a slot's target position (its own cell, or where the row says it would be)
        local function targetCenter(g)
            local f = g.frame
            if f and f.Parent and f.AbsoluteSize.X > 4 then return cellCenter(f) end
            if rowRef and rowRef.cell.Parent then
                local ref = rowRef.cell
                local abs = Vector2.new(ref.AbsolutePosition.X + (g.pos - rowRef.pos) * rowPitch, ref.AbsolutePosition.Y)
                return toMouse(abs + ref.AbsoluteSize / 2, ref)
            end
            return nil
        end

        -- is this cell holding a weapon that belongs in `slot`? (icon first, then any name it exposes)
        local function isBrassName(n)
            n = tostring(n):lower()
            return n:find("knuckle", 1, true) ~= nil or n:find("brass", 1, true) ~= nil
        end
        local function cellIsItemFor(cell, slot)
            local tmp = { frame = cell, slot = slot }
            local exp = expected[slot]
            local imgs = readSlotImages(tmp)
            -- slot 8 takes the FIST only. Brass Knuckles are never moved into it: they get equipped instead
            -- (the Fist then uses them), see equipBrass below.
            if slot == "8" then
                for id in pairs(imgs or {}) do
                    if toolByIcon[id] and isBrassName(toolByIcon[id]) then return false end
                end
                for _, str in ipairs(harvestSlotStrings(tmp) or {}) do
                    if isBrassName(str) then return false end
                end
            end
            if exp and imgs then
                for id in pairs(imgs) do if exp.images[id] then return true end end
            end
            local strs = harvestSlotStrings(tmp) or {}
            for id in pairs(imgs or {}) do
                local tn = toolByIcon[id]
                if tn then strs[#strs + 1] = tn end
            end
            for _, str in ipairs(strs) do
                if closeToSlot(slot, str) then return true end
            end
            return false
        end

        -- the item cells of the open inventory screen (outermost square-ish cells that hold a picture or a name)
        local function inventoryCells()
            local containers, seenC = {}, {}
            local function addC(o)
                if o and o.Parent and not seenC[o] and chainShown(o) then
                    seenC[o] = true
                    containers[#containers + 1] = o
                end
            end
            addC(invWatchDefault)
            addC(invWatchGrid)
            local cells = {}
            for _, root in ipairs(containers) do
                for _, d in ipairs(root:GetDescendants()) do
                    if d:IsA("GuiObject") and d.Visible and not d:IsDescendantOf(sg) then
                        local sz = d.AbsoluteSize
                        if sz.X >= 24 and sz.X <= 260 and sz.Y >= 24 and sz.Y <= 260
                            and sz.X / sz.Y >= 0.6 and sz.X / sz.Y <= 1.6 and chainShown(d) then
                            local inside = false
                            for _, c in ipairs(cells) do
                                if d == c or d:IsDescendantOf(c) then inside = true break end
                            end
                            if not inside then cells[#cells + 1] = d end
                        end
                    end
                end
            end
            return cells
        end

        -- a slot needs fixing when its weapon is in your inventory but the slot doesn't show it
        -- (wrong thing in it, OR empty: an empty hotbar slot reads "unknown", so it counts too)
        local function needsFix(g)
            if (g.slot == "2" or g.slot == "3") and not AutoAccessoryEnabled then return false end
            local exp = expected[g.slot]
            return exp ~= nil and exp.found and g.status ~= "ok"
        end

        -- If the inventory screen's frame wasn't recognised, look at every picture-holding cell above the hotbar
        local function looseCells()
            local cells = {}
            local hotParent = rowRef and rowRef.cell and rowRef.cell.Parent
            local hotTop = (rowRef and rowRef.cell and rowRef.cell.Parent) and rowRef.cell.AbsolutePosition.Y - 8 or math.huge
            local roots = { p:FindFirstChildOfClass("PlayerGui") }
            for _, root in ipairs(roots) do
                for _, d in ipairs(root:GetDescendants()) do
                    if d:IsA("GuiObject") and d.Visible and not d:IsDescendantOf(sg)
                        and not (hotParent and d:IsDescendantOf(hotParent)) then
                        local sz = d.AbsoluteSize
                        if sz.X >= 24 and sz.X <= 200 and sz.Y >= 24 and sz.Y <= 200
                            and sz.X / sz.Y >= 0.6 and sz.X / sz.Y <= 1.6
                            and d.AbsolutePosition.Y + sz.Y <= hotTop and chainShown(d) then
                            local imgs = readSlotImages({ frame = d, slot = "" })
                            if imgs and next(imgs) then
                                local inside = false
                                for _, c in ipairs(cells) do
                                    if d == c or d:IsDescendantOf(c) then inside = true break end
                                end
                                if not inside then cells[#cells + 1] = d end
                            end
                        end
                    end
                end
            end
            return cells
        end

        -- Last resort for finding a weapon on the open backpack screen: look for its picture (or its name) on
        -- ANY visible image/label outside the hotbar, no matter how the backpack's frames are arranged.
        local function iconSource(slot)
            local exp = expected[slot]
            local pg = p:FindFirstChildOfClass("PlayerGui")
            if not (exp and pg) then return nil end
            local hotParent = rowRef and rowRef.cell and rowRef.cell.Parent
            for _, d in ipairs(pg:GetDescendants()) do
                if (d:IsA("ImageLabel") or d:IsA("ImageButton")) and d.Visible and not d:IsDescendantOf(sg)
                    and not (hotParent and d:IsDescendantOf(hotParent)) then
                    local id = imageId(d.Image)
                    local sz = d.AbsoluteSize
                    if id and exp.images[id] and sz.X >= 16 and sz.Y >= 16 and chainShown(d) then
                        return d
                    end
                end
            end
            for _, d in ipairs(pg:GetDescendants()) do
                if (d:IsA("TextLabel") or d:IsA("TextButton")) and d.Visible and not d:IsDescendantOf(sg)
                    and not (hotParent and d:IsDescendantOf(hotParent)) and #d.Text > 2 and chainShown(d)
                    and closeToSlot(slot, d.Text) then
                    local par = d.Parent
                    if par and par:IsA("GuiObject") and par.AbsoluteSize.X <= 260 then return par end
                    return d
                end
            end
            return nil
        end

        -- does this backpack/hotbar cell show this particular tool?
        local function cellMatchesTool(cell, tool)
            local tid = imageId(tool.TextureId or "")
            local imgs = readSlotImages({ frame = cell, slot = "" })
            if tid and imgs and imgs[tid] then return true end
            local want = nz(tool.Name)
            -- longer names of OTHER tools you own that contain this one ("spikedkittystanley" contains "kittystanley")
            local longer = {}
            for _, o in ipairs(PLAN.allTools) do
                if o ~= tool then
                    local on = nz(o.Name)
                    if #on > #want and on:find(want, 1, true) then longer[#longer + 1] = on end
                end
            end
            for _, str in ipairs(harvestSlotStrings({ frame = cell }) or {}) do
                local n = nz(str)
                if #n >= 3 then
                    if n == want then return true end
                    if n:find(want, 1, true) then
                        local other = false
                        for _, on in ipairs(longer) do
                            if n:find(on, 1, true) then other = true break end
                        end
                        if not other then return true end
                    end
                end
            end
            return false
        end

        -- true when what sits in this accessory slot is NOT an accessory weapon (a Phone, a Spray, the
        -- Loveboard, a high-DPS weapon, or anything the script doesn't recognise at all)
        accessoryForeign = function(g)
            if not (g.frame and g.frame.Parent) then return false end
            for _, tool in ipairs(PLAN.allTools) do
                if cellMatchesTool(g.frame, tool) then
                    local e = WEAPONS.tool(tool)
                    return not (e ~= nil and e.kind == "acc")
                end
            end
            return true -- matches none of your tools: unknown, so it doesn't belong
        end

        -- work out the next fix: returns slot entry, the cell to pick up
        local function planFix()
            local gridCells = autoFix.grid
            for _, g in ipairs(glows) do
                if needsFix(g) and (autoFix.tries[g.slot] or 0) < MAXTRIES then
                    if not gridCells then
                        gridCells = inventoryCells()
                        if #gridCells == 0 then gridCells = looseCells() end
                        autoFix.grid = gridCells -- reused for the rest of this pass (cleared at its start)
                    end
                    if g.slot == "2" or g.slot == "3" then
                        -- accessory slot: your best accessory weapon that isn't already on the hotbar
                        for _, tool in ipairs(PLAN.accessory) do
                            local inAccSlot, wrongBar = false, nil
                            for pos, hc in pairs(hotCellsByPos) do
                                if hc.Parent and cellMatchesTool(hc, tool) then
                                    if pos == 2 or pos == 3 then inAccSlot = true else wrongBar = hc end
                                end
                            end
                            if not inAccSlot then
                                if wrongBar and wrongBar ~= g.frame then return g, wrongBar end -- e.g. a purse in slot 7
                                for _, c in ipairs(gridCells) do
                                    if c ~= g.frame and cellMatchesTool(c, tool) then return g, c end
                                end
                            end
                        end
                    else
                        for _, c in ipairs(gridCells) do
                            if c ~= g.frame and cellIsItemFor(c, g.slot) then return g, c end
                        end
                        for pos, c in pairs(hotCellsByPos) do
                            if c ~= g.frame and c.Parent then
                                local owner
                                for _, o in ipairs(glows) do if o.pos == pos then owner = o break end end
                                if not (owner and owner.status == "ok") and cellIsItemFor(c, g.slot) then
                                    return g, c
                                end
                            end
                        end
                        local d = iconSource(g.slot)
                        if d and d ~= g.frame then return g, d end
                    end
                end
            end
            return nil
        end

        ------------------------------------------------------------
        -- Wrong item in a slot with nothing better to put there (for example the Brass Knuckles sitting in slot 1,
        -- or a random tool in slot 4): it's moved out of the way, into an empty accessory slot (2 or 3).
        ------------------------------------------------------------
        local EVICT_SLOTS = { ["1"] = true, ["4"] = true, ["5"] = true, ["6"] = true, ["7"] = true, ["8"] = true, ["9"] = true, ["0"] = true }
        local function slotFilled(g)
            local name = readSlotName(g)
            if name and name ~= "" then return true end
            local imgs = readSlotImages(g)
            return imgs ~= nil and next(imgs) ~= nil
        end
        local function slotItemName(g)
            local name = readSlotName(g)
            if name and name ~= "" then return (tostring(name):gsub("^%d+%s*", "")) end
            for id in pairs(readSlotImages(g) or {}) do
                if toolByIcon[id] then return toolByIcon[id] end
            end
            return "an item"
        end
        -- returns the slot entry whose item should be taken out of the hotbar and put back in the backpack:
        --  a wrong item in slots 1/4/5/8/0 with nothing better to put there, or a non-accessory weapon in slot 2/3
        local function planEvict()
            if not AutoFixEnabled then return nil end
            for _, g in ipairs(glows) do
                local isAcc = (g.slot == "2" or g.slot == "3")
                if (EVICT_SLOTS[g.slot] or isAcc) and (autoFix.tries["e" .. g.slot] or 0) < 2 and g.frame and g.frame.Parent then
                    if isAcc then
                        if slotFilled(g) and accessoryForeign and accessoryForeign(g) and not needsFix(g) then return g end
                    elseif slotFilled(g) then
                        local holdsGear = false
                        for _, tool in ipairs(PLAN.equipAll) do
                            if cellMatchesTool(g.frame, tool) then holdsGear = true break end
                        end
                        -- right item (by name or picture) = leave it. Anything else, known or not, goes back.
                        if not holdsGear and not cellIsItemFor(g.frame, g.slot) then
                            if g.slot == "6" or g.slot == "7" or g.slot == "9" then
                                -- Loveboard / Snowball Launcher skins carry their own names, so in these three slots only
                                -- things positively known to belong elsewhere are moved
                                for _, tool in ipairs(PLAN.owned) do
                                    if cellMatchesTool(g.frame, tool) then return g end
                                end
                            else
                                return g
                            end
                        end
                    end
                end
            end
            return nil
        end

        -- true while you've told it to "fix now" (tap the prompt): it then ignores the idle / fight checks
        autoFix.forced = function() return os.clock() < autoFix.forceUntil end

        -- would a click at the centre of this cell really land on it? (checked against what's on screen right now)
        autoFix.hits = function(pt, cell)
            local pg = p:FindFirstChildOfClass("PlayerGui")
            if not (pg and cell and cell.Parent and cell:IsDescendantOf(pg)) then return true end -- can't check: allow
            local a = autoFix.addInset
            if a == nil then a = probeInset() end
            local inset = a and GuiService_:GetGuiInset() or Vector2.new(0, 0)
            local okq, list = pcall(function() return pg:GetGuiObjectsAtPosition(pt.X - inset.X, pt.Y - inset.Y) end)
            if not okq then return true end
            for _, o in ipairs(list) do
                if o == cell or o:IsDescendantOf(cell) then return true end
            end
            return false
        end
        -- the click point for a cell, verified. If it would miss (screen shifted, offset wrong), the offset is
        -- re-measured once; if it still misses, the move is skipped instead of clicking somewhere random.
        autoFix.safePoint = function(cell)
            if not (cell and cell.Parent) then return nil end
            local sz = cell.AbsoluteSize
            if sz.X < 4 or sz.Y < 4 then return nil end
            local pt = cellCenter(cell)
            local cam = workspace.CurrentCamera
            local function inScreen(v)
                return not cam or (v.X >= 0 and v.Y >= 0 and v.X <= cam.ViewportSize.X and v.Y <= cam.ViewportSize.Y)
            end
            if inScreen(pt) and autoFix.hits(pt, cell) then return pt end
            autoFix.addInset, autoFix.calibrated = nil, false
            AutoFixMemory.addInset = nil
            calibrateOffset(cell)
            pt = cellCenter(cell)
            if inScreen(pt) and autoFix.hits(pt, cell) then return pt end
            afLog("the click would have missed", cell.Name, "- skipped that move")
            return nil
        end

        local function release(pt)
            pcall(function() VIM:SendMouseButtonEvent(pt.X, pt.Y, 0, false, game, 0) end)
        end

        -- Six ways to move an item, fastest first. The script tries them in turn, remembers which one the
        -- game responds to, and uses that one first from then on (even if the script is re-run):
        --  1 = fast mouse drag   2 = standard mouse drag   3 = slow mouse drag   4 = mouse: tap item, tap slot
        --  5 = finger drag       6 = finger: tap item, tap slot
        local METHODS = 6
        local TOUCH_FIRST = game:GetService("UserInputService").TouchEnabled
            and not game:GetService("UserInputService").MouseEnabled
        local function moveWith(method, a, b)
            local ok = pcall(function()
                if method >= 5 then
                    local id = 11
                    VIM:SendTouchEvent(id, 0, a.X, a.Y) -- finger down on the item
                    task.wait(0.06)
                    if method == 5 then
                        task.wait(0.2) -- hold, so it's treated as a drag and not a tap
                        for i = 1, 10 do
                            local t = i / 10
                            VIM:SendTouchEvent(id, 1, a.X + (b.X - a.X) * t, a.Y + (b.Y - a.Y) * t)
                            task.wait(0.02)
                        end
                        task.wait(0.1)
                        VIM:SendTouchEvent(id, 2, b.X, b.Y) -- finger up on the slot
                    else
                        VIM:SendTouchEvent(id, 2, a.X, a.Y)
                        task.wait(0.2)
                        VIM:SendTouchEvent(id, 0, b.X, b.Y)
                        task.wait(0.06)
                        VIM:SendTouchEvent(id, 2, b.X, b.Y)
                    end
                    return
                end
                VIM:SendMouseMoveEvent(a.X, a.Y, game)
                task.wait(0.04)
                if method == 4 then
                    VIM:SendMouseButtonEvent(a.X, a.Y, 0, true, game, 0)
                    task.wait(0.05)
                    VIM:SendMouseButtonEvent(a.X, a.Y, 0, false, game, 0)
                    task.wait(0.2)
                    VIM:SendMouseMoveEvent(b.X, b.Y, game)
                    task.wait(0.1)
                    VIM:SendMouseButtonEvent(b.X, b.Y, 0, true, game, 0)
                    task.wait(0.05)
                    VIM:SendMouseButtonEvent(b.X, b.Y, 0, false, game, 0)
                    return
                end
                local hold, steps, stepWait, hover
                if method == 1 then hold, steps, stepWait, hover = 0.03, 3, 0.004, 0.02
                elseif method == 2 then hold, steps, stepWait, hover = 0.28, 12, 0.025, 0.15
                else hold, steps, stepWait, hover = 0.55, 28, 0.035, 0.35 end
                VIM:SendMouseButtonEvent(a.X, a.Y, 0, true, game, 0)
                task.wait(hold) -- hold so the backpack treats it as a drag
                -- a tiny wiggle first: many backpacks only start a drag after the pointer has moved a bit
                VIM:SendMouseMoveEvent(a.X + 4, a.Y + 4, game)
                task.wait(0.015)
                for i = 1, steps do
                    local t = i / steps
                    VIM:SendMouseMoveEvent(a.X + (b.X - a.X) * t, a.Y + (b.Y - a.Y) * t, game)
                    task.wait(stepWait)
                end
                task.wait(hover) -- hover over the slot before letting go
                VIM:SendMouseButtonEvent(b.X, b.Y, 0, false, game, 0)
            end)
            if not ok then release(b) end -- never leave the virtual button held down
            return ok
        end

        ------------------------------------------------------------
        -- Hands-free mode: with the backpack CLOSED, it opens the backpack by itself (the game's own
        -- toggle key), drags the weapon into place, then closes it again. Only while you're idle.
        ------------------------------------------------------------
        local TOGGLE_KEYS = { Enum.KeyCode.H, Enum.KeyCode.Backquote } -- the game's "Close H"; Roblox's default is `

        local function searchBoxVisible() -- fallback "is the inventory screen up?" when its frame isn't known yet
            local pg = p:FindFirstChildOfClass("PlayerGui")
            if not pg then return false end
            for _, d in ipairs(pg:GetDescendants()) do
                if (d:IsA("TextLabel") or d:IsA("TextBox") or d:IsA("TextButton")) and not d:IsDescendantOf(sg) then
                    local ph = d:IsA("TextBox") and d.PlaceholderText or ""
                    if (d.Text == "Search" or ph == "Search") and chainShown(d) then return true end
                end
            end
            return false
        end

        local function inventoryIsOpen()
            local known = false
            if invWatchDefault and invWatchDefault.Parent then
                known = true
                if chainShown(invWatchDefault) then return true end
            end
            if invWatchGrid and invWatchGrid.Parent then
                known = true
                if chainShown(invWatchGrid) then return true end
            end
            if known then return false end
            return searchBoxVisible()
        end

        local function tapKey(code)
            SynthDragging = true
            pcall(function()
                VIM:SendKeyEvent(true, code, false, game)
                task.wait(0.04)
                VIM:SendKeyEvent(false, code, false, game)
            end)
            task.wait(0.08)
            SynthDragging = false
        end

        local function waitForOpenState(wantOpen, timeout)
            local t0 = os.clock()
            while os.clock() - t0 < timeout do
                if inventoryIsOpen() == wantOpen then return true end
                task.wait(0.05)
            end
            return inventoryIsOpen() == wantOpen
        end

        local function openInventory()
            if inventoryIsOpen() then return true end
            local order = {}
            if autoFix.key then order[1] = autoFix.key end
            for _, k in ipairs(TOGGLE_KEYS) do if k ~= autoFix.key then order[#order + 1] = k end end
            for _, k in ipairs(order) do
                tapKey(k)
                if waitForOpenState(true, 1.3) then
                    autoFix.key = k -- remember which key worked
                    return true
                end
            end
            return false
        end

        local function closeInventory()
            if not inventoryIsOpen() or not autoFix.key then return end
            tapKey(autoFix.key)
            if waitForOpenState(false, 1.3) then return end
            tapKey(autoFix.key) -- one more try, then leave it be
            waitForOpenState(false, 1.0)
        end

        local function rescanAll()
            pcall(resolveSlotFrames)
            pcall(scanInventory)
            for _, g2 in ipairs(glows) do pcall(checkSlot, g2) end
        end

        ------------------------------------------------------------
        -- Equippables (knuckles/gloves, boots, cloak, crown): they are NOT moved into a slot. Each is clicked
        -- in the backpack, then the empty screen is clicked ("click off"), then the item is put back out of your
        -- hands so it stays equipped. Once per spawn. (The Fist, in slot 8, uses the knuckles.)
        ------------------------------------------------------------
        -- Listens to the game's console output (LogService), so the script can hear when the game itself says
        -- something about equipping. Anything mentioning equip / knuckles / brass is remembered (and echoed as
        -- "[AutoFix] heard: ..." so you can see exactly what your game prints).
        local heard = {}
        game:GetService("LogService").MessageOut:Connect(function(msg)
            if type(msg) ~= "string" then return end
            if msg:find("[AutoFix]", 1, true) or msg:find("[Exotica]", 1, true) then return end
            local l = msg:lower()
            if l:find("equip", 1, true) or l:find("knuckle", 1, true) or l:find("brass", 1, true) or l:find("gloves", 1, true) then
                heard[#heard + 1] = { t = os.clock(), l = l }
                if #heard > 30 then table.remove(heard, 1) end
                warn("[AutoFix] heard:", msg:sub(1, 140))
            end
        end)

        -- true when something shows this piece of gear is on you: in your hands, a matching part on your
        -- character, an attribute naming it, the game saying "equipped", or (backpack open) an "Equipped" mark
        local function isGearEquipped(tool, cell)
            local ch = p.Character
            if ch and tool.Parent == ch then return true end
            local want = nz(tool.Name)
            local words = {}
            for w in tostring(tool.Name):lower():gmatch("%a+") do if #w >= 4 then words[#words + 1] = w end end
            if ch then
                for _, d in ipairs(ch:GetDescendants()) do
                    if not d:IsA("Tool") and not d:IsDescendantOf(tool) then
                        local n = nz(d.Name)
                        if #n >= 6 and (n == want or n:find(want, 1, true) or want:find(n, 1, true)) then return true end
                    end
                end
                for _, holder in ipairs({ ch, p }) do
                    for k, v in pairs(holder:GetAttributes()) do
                        local kk, vv = nz(k), (type(v) == "string") and nz(v) or ""
                        if kk:find(want, 1, true) or (#vv >= 4 and vv:find(want, 1, true)) then
                            if v ~= false and v ~= 0 and v ~= "" then return true end
                        end
                    end
                end
            end
            local state
            for _, h in ipairs(heard) do
                if h.t >= (autoFix.spawnAt or 0) then
                    local hit = h.l:find(tostring(tool.Name):lower(), 1, true)
                    if not hit then for _, w in ipairs(words) do if h.l:find(w, 1, true) then hit = true break end end end
                    if hit then
                        if h.l:find("unequip", 1, true) or h.l:find("de-equip", 1, true) or h.l:find("not equipped", 1, true) then state = false
                        elseif h.l:find("equip", 1, true) then state = true end
                    end
                end
            end
            if state ~= nil then return state end
            if cell then
                for _, d in ipairs(cell:GetDescendants()) do
                    if (d:IsA("TextLabel") or d:IsA("TextButton")) and d.Visible and tostring(d.Text):lower():find("equipped", 1, true)
                        and not tostring(d.Text):lower():find("un", 1, true) then
                        return true
                    end
                end
            end
            return false
        end

        local function equipQueue()
            local list = {}
            if not AutoEquipEnabled then return list end
            local ch = p.Character
            for _, t in ipairs(PLAN.equip) do
                if t.Parent and not autoFix.equipDone[t.Name] and (autoFix.equipTries[t.Name] or 0) < 2
                    and not (ch and t.Parent == ch) and not isGearEquipped(t) then
                    list[#list + 1] = t
                end
            end
            return list
        end
        local function brassNeeded() return #equipQueue() > 0 end

        local function tapAt(pt)
            SynthDragging = true
            pcall(function()
                if (autoFix.method or 0) >= 5 then
                    VIM:SendTouchEvent(12, 0, pt.X, pt.Y)
                    task.wait(0.06)
                    VIM:SendTouchEvent(12, 2, pt.X, pt.Y)
                else
                    VIM:SendMouseMoveEvent(pt.X, pt.Y, game)
                    task.wait(0.05)
                    VIM:SendMouseButtonEvent(pt.X, pt.Y, 0, true, game, 0)
                    task.wait(0.05)
                    VIM:SendMouseButtonEvent(pt.X, pt.Y, 0, false, game, 0)
                end
            end)
            task.wait(0.04)
            SynthDragging = false
        end

        -- a spot on the screen that is outside the backpack and not on any button, so a click there only "clicks off"
        local function freeScreenPoint()
            local cam = workspace.CurrentCamera
            if not cam then return nil end
            local size = cam.ViewportSize
            local inset = GuiService_:GetGuiInset()
            local pg = p:FindFirstChildOfClass("PlayerGui")
            local panel = invWatchDefault and invWatchDefault.Parent and invWatchDefault or nil
            for _, f in ipairs({ { 0.5, 0.07 }, { 0.35, 0.07 }, { 0.65, 0.07 }, { 0.5, 0.14 }, { 0.12, 0.4 }, { 0.88, 0.4 } }) do
                local mx, my = size.X * f[1], size.Y * f[2]
                local gx, gy = mx, my
                if autoFix.addInset then gx, gy = mx - inset.X, my - inset.Y end
                local free = true
                local okq = pcall(function()
                    local function bad(list)
                        for _, o in ipairs(list) do
                            if not o:IsDescendantOf(sg) and (o:IsA("GuiButton") or o:IsA("TextBox") or o:IsA("ScrollingFrame") or o.Active) then
                                return true
                            end
                        end
                        return false
                    end
                    if pg and bad(pg:GetGuiObjectsAtPosition(gx, gy)) then free = false end
                    if panel and chainShown(panel) then
                        local ap, az = panel.AbsolutePosition, panel.AbsoluteSize
                        if gx >= ap.X and gx <= ap.X + az.X and gy >= ap.Y and gy <= ap.Y + az.Y then free = false end
                    end
                end)
                if okq and free then return Vector2.new(mx, my) end
            end
            return nil
        end

        -- the hotbar cell currently showing this tool, if it is on your hotbar
        local function hotbarCellFor(tool)
            for _, c in pairs(hotCellsByPos) do
                if c.Parent and c.AbsoluteSize.X > 4 and cellMatchesTool(c, tool) then return c end
            end
            return nil
        end

        -- Equippables (knuckles, gloves, boots, cloak, crown) don't belong on the hotbar: they get clicked on to
        -- equip and then live in the backpack. This finds one that's sitting on the hotbar.
        local function planEquipEvict()
            if not (AutoFixEnabled and AutoEquipEnabled) then return nil end
            for _, tool in ipairs(PLAN.equipAll) do
                if (autoFix.tries["x" .. tool.Name] or 0) < 2 and tool.Parent ~= p.Character then
                    local c = hotbarCellFor(tool)
                    if c then return tool, c end
                end
            end
            return nil
        end

        -- a spot on the open backpack screen that is empty (not on any item), to drop an item onto
        local function emptyInventorySpot(cells)
            local container
            if cells and cells[1] then
                container = cells[1]:FindFirstAncestorWhichIsA("ScrollingFrame") or cells[1].Parent
            end
            if not container and invWatchDefault and invWatchDefault.Parent and chainShown(invWatchDefault) then
                container = invWatchDefault
            end
            if not container then return nil end
            local ap, az = container.AbsolutePosition, container.AbsoluteSize
            for _, fy in ipairs({ 0.9, 0.7, 0.5, 0.3, 0.12 }) do
                for _, fx in ipairs({ 0.1, 0.3, 0.5, 0.7, 0.9 }) do
                    local px, py = ap.X + az.X * fx, ap.Y + az.Y * fy
                    local clear = true
                    for _, c in ipairs(cells or {}) do
                        local cp, cs = c.AbsolutePosition, c.AbsoluteSize
                        if px >= cp.X - 6 and px <= cp.X + cs.X + 6 and py >= cp.Y - 6 and py <= cp.Y + cs.Y + 6 then
                            clear = false
                            break
                        end
                    end
                    if clear then return toMouse(Vector2.new(px, py), container) end
                end
            end
            return nil
        end

        -- Some gear (for example the Gorilla Gel) is not a tool at all: it only exists as an item in the backpack screen.
        -- When the backpack is open, any item whose name matches these words is clicked once per spawn.
        local GEAR_WORDS = { "gorilla", "hairglue", "hairgel", "snure" }
        autoFix.guiGear = function()
            local cells = inventoryCells()
            if #cells == 0 then cells = looseCells() end
            for _, c in ipairs(cells) do
                for _, str in ipairs(harvestSlotStrings({ frame = c }) or {}) do
                    local n = nz(str)
                    for _, w in ipairs(GEAR_WORDS) do
                        local key = "gui:" .. w
                        if n:find(w, 1, true) and not autoFix.equipDone[key] then
                            autoFix.equipDone[key] = true
                            AutoFixMemory.guiGear = true -- next time, open the backpack at spawn to equip it
                            calibrateOffset(c)
                            local pt = autoFix.safePoint(c)
                            if pt then
                                tapAt(pt)
                                task.wait(0.08)
                                local away = freeScreenPoint()
                                if not away then
                                    local cam = workspace.CurrentCamera
                                    if cam then away = Vector2.new(cam.ViewportSize.X * 0.5, cam.ViewportSize.Y * 0.08) end
                                end
                                if away then
                                    local freed = freeClickPath(away.X, away.Y, autoFix.addInset ~= false)
                                    tapAt(away)
                                    restoreClickPath(freed)
                                end
                                task.wait(0.06)
                                afLog("equipped backpack item:", str)
                                return str
                            end
                        end
                    end
                end
            end
            return nil
        end

        local function equipGear(tool)
            autoFix.equipTries[tool.Name] = (autoFix.equipTries[tool.Name] or 0) + 1
            if isGearEquipped(tool) then autoFix.equipDone[tool.Name] = true return true end
            -- it may be sitting on your hotbar (the game drops new gear into the first free slot) or in the
            -- backpack screen. The hotbar is always visible, so try it first.
            local cell = hotbarCellFor(tool)
            local where = "hotbar"
            if not cell then
                where = "backpack"
                local cells = inventoryCells()
                if #cells == 0 then cells = looseCells() end
                for _, c in ipairs(cells) do
                    if cellMatchesTool(c, tool) then cell = c break end
                end
            end
            if not cell then afLog(tool.Name .. ": not found on the hotbar or in the backpack screen") return false end
            afLog(tool.Name .. ": found on the " .. where)
            calibrateOffset(cell)
            local tapPt = autoFix.safePoint(cell)
            if not tapPt then afLog(tool.Name, ": couldn't aim a safe click at it, skipped") return false end
            tapAt(tapPt) -- click the item
            task.wait(0.08)
            -- click off onto the empty screen. If a button or panel is on that spot, the click goes through it
            -- to what's behind it (the same click-through the aim assist uses), so it can never press a UI.
            local away = freeScreenPoint()
            if not away then
                local cam = workspace.CurrentCamera
                if cam then away = Vector2.new(cam.ViewportSize.X * 0.5, cam.ViewportSize.Y * 0.08) end
            end
            if away then
                local freed = freeClickPath(away.X, away.Y, autoFix.addInset ~= false)
                tapAt(away)
                restoreClickPath(freed)
            end
            task.wait(0.1)
            if tool.Parent ~= p.Character then -- the click didn't equip it: use Roblox's own equip call
                pcall(function()
                    local hum = p.Character and p.Character:FindFirstChildOfClass("Humanoid")
                    if hum then hum:EquipTool(tool) end
                end)
                task.wait(0.06)
            end
            local inHand = tool.Parent == p.Character
            if inHand then
                -- it's on now; put it back out of your hands so it stays equipped
                pcall(function()
                    local hum = p.Character and p.Character:FindFirstChildOfClass("Humanoid")
                    if hum then hum:UnequipTools() end
                end)
                task.wait(0.04)
            end
            local ok = inHand or isGearEquipped(tool, cell)
            if ok then autoFix.equipDone[tool.Name] = true end
            afLog(tool.Name, "| was in hand:", inHand, "| counts as equipped:", ok, "| back out of hand:", tool.Parent ~= p.Character)
            return ok
        end

        local function checkAll() for _, g2 in ipairs(glows) do pcall(checkSlot, g2) end end

        local function runAutoFix(handsFree, equipOnly)
            autoFix.busy = true
            local moved, failed, extra = {}, {}, {}
            local opened, weOpened = false, false
            autoFix.grid = nil
            pcall(function()
                if handsFree then
                    local n = #equipQueue()
                    if not equipOnly then
                        for _, g in ipairs(glows) do if needsFix(g) then n = n + 1 end end
                        if planEvict() then n = n + 1 end
                    end
                    autoFix.closedCycles = autoFix.closedCycles + 1
                    autoFix.cool = os.clock() + 20
                    if autoFixStatus then pcall(autoFixStatus, n, equipOnly) end
                    autoFix.silent = true -- no backpack hum for this quick visit
                    afLog("opening the backpack, things to do:", n)
                    pcall(resolveSlotFrames) -- fresh hotbar cells (your GUI is rebuilt on every spawn)
                    local onHotbar = false -- gear has to come off the hotbar, which needs the backpack open
                    if false then
                        for _, t in ipairs(equipQueue()) do
                            if not hotbarCellFor(t) then onHotbar = false break end
                        end
                    end
                    if onHotbar then
                        opened = true -- nothing to open: your gear is on the hotbar already
                        afLog("gear is on the hotbar, no need to open the backpack")
                    else
                        local wasOpen = inventoryIsOpen()
                        if not openInventory() then
                            afLog("could not open the backpack with H or `")
                            autoFix.cool = os.clock() + 120
                            return
                        end
                        opened, weOpened = true, not wasOpen
                        task.wait(0.15) -- let it draw, then read it fresh
                        rescanAll()
                    end
                end
                -- equippables don't belong on the hotbar: drag them back into the backpack first
                for _ = 1, 6 do
                    if not (AutoFixEnabled and AutoEquipEnabled) then break end
                    if not (opened or autoOpen) then break end
                    if not equipOnly and not autoFix.forced() and (os.clock() - LastUserInput < 0.25 or Engine.run ~= nil) then break end
                    local tool, hcell = planEquipEvict()
                    if not tool then break end
                    autoFix.grid = nil
                    local cells = inventoryCells()
                    if #cells == 0 then cells = looseCells() end
                    local spot = emptyInventorySpot(cells)
                    local key = "x" .. tool.Name
                    autoFix.tries[key] = (autoFix.tries[key] or 0) + 1
                    if not spot then afLog("no empty spot in the backpack to drop", tool.Name, "on") break end
                    calibrateOffset(hcell)
                    local src = autoFix.safePoint(hcell)
                    if not src then break end
                    local method = ((autoFix.method or (TOUCH_FIRST and 5 or 1)) - 1 + autoFix.tries[key] - 1) % METHODS + 1
                    SynthDragging = true
                    moveWith(method, src, spot)
                    task.wait(0.05)
                    SynthDragging = false
                    task.wait(0.05)
                    pcall(resolveSlotFrames)
                    if not hotbarCellFor(tool) then
                        extra[#extra + 1] = "Moved " .. tool.Name .. " off your hotbar."
                        playEmbedded("confirm", 0.6, 0.9, 1)
                        afLog(tool.Name, "moved off the hotbar (method", method .. ")")
                    else
                        afLog(tool.Name, "is still on the hotbar after method", method)
                    end
                end
                autoFix.grid = nil
                for _ = 1, (equipOnly and 0 or 12) do
                    if not AutoFixEnabled then break end
                    if not (opened or autoOpen) then break end
                    if not autoFix.forced() and os.clock() - LastUserInput < 0.25 then break end -- you touched something just now: step back
                    if Engine.run ~= nil then break end
                    local g, cell = planFix()
                    local evictDst, evictName
                    if not g then
                        g = planEvict()
                        if not g then break end
                        cell = g.frame
                        evictName = slotItemName(g)
                        local ecells = autoFix.grid or inventoryCells()
                        if #ecells == 0 then ecells = looseCells() end
                        evictDst = emptyInventorySpot(ecells)
                        if not evictDst then
                            afLog("no empty spot in the backpack to drop", evictName)
                            autoFix.tries["e" .. g.slot] = 2
                            break
                        end
                    end
                    calibrateOffset(cell)
                    local src, dst = autoFix.safePoint(cell), evictDst or targetCenter(g)
                    if not (src and dst) then afLog("no safe screen position for slot", g.slot) break end
                    if g.frame and not evictDst and g.frame.Parent and not autoFix.hits(dst, g.frame) then
                        dst = autoFix.safePoint(g.frame)
                        if not dst then afLog("the drop point for slot", g.slot, "would have missed, skipped") break end
                    end
                    local tkey = evictDst and ("e" .. g.slot) or g.slot
                    local tries = (autoFix.tries[tkey] or 0) + 1
                    autoFix.tries[tkey] = tries
                    local startMethod = autoFix.method or (TOUCH_FIRST and 5 or 1)
                    local method = ((startMethod - 1 + tries - 1) % METHODS) + 1
                    afLog(string.format("slot %s (%s): method %d, from %s (%d,%d, %dx%d) to (%d,%d)", g.slot, tostring(g.status), method,
                        cell:GetFullName():sub(-60), src.X, src.Y, cell.AbsoluteSize.X, cell.AbsoluteSize.Y, dst.X, dst.Y))
                    -- the press/release watchers are only for diagnosing; skipped once a method is known to work
                    local watching = autoFix.method == nil
                    local srcRec = watching and watchHits(cell) or { began = 0, ended = 0, entered = false, stop = function() end }
                    local dstRec = watching and g.frame and watchHits(g.frame) or nil
                    SynthDragging = true
                    moveWith(method, src, dst)
                    task.wait(0.05)
                    SynthDragging = false
                    if watching then afLog(string.format("  item: pointer entered=%s, press landed=%d, release landed=%d | slot: entered=%s, press=%d, release=%d",
                        tostring(srcRec.entered), srcRec.began, srcRec.ended,
                        tostring(dstRec and dstRec.entered), dstRec and dstRec.began or 0, dstRec and dstRec.ended or 0)) end
                    srcRec.stop()
                    if dstRec then dstRec.stop() end
                    task.wait(0.03)
                    checkAll() -- did it land?
                    if evictDst then
                        if not slotFilled(g) then
                            autoFix.method = method
                            AutoFixMemory.method = method
                            extra[#extra + 1] = "Moved " .. evictName .. " out of slot " .. g.slot .. " and back to your inventory."
                            playEmbedded("confirm", 0.6, 0.9, 1)
                            afLog("slot", g.slot, "cleared of", evictName)
                        end
                    elseif g.status == "ok" then
                        autoFix.method = method -- the game responds to this one, use it first from now on
                        AutoFixMemory.method = method
                        moved[#moved + 1] = g
                        g.flashUntil = os.clock() + 3.2 -- the slot flashes on screen (see the glow loop)
                        playEmbedded("confirm", 0.7, 0.92 + 0.1 * #moved, 1)
                        afLog("slot", g.slot, "fixed with method", method)
                    end
                end
                -- equippables, once per spawn (knuckles/gloves, boots, cloak, crown)
                for _ = 1, 5 do
                    if not (opened or autoOpen) then break end
                    -- right after a respawn (equipOnly) it equips no matter what you're doing
                    if not equipOnly and not autoFix.forced() and (os.clock() - LastUserInput < 0.25 or Engine.run ~= nil) then break end
                    local q = equipQueue()
                    if #q == 0 then break end
                    if equipGear(q[1]) then
                        extra[#extra + 1] = "Equipped your " .. q[1].Name .. "."
                    else
                        failed[#failed + 1] = "Couldn't equip " .. q[1].Name .. " yet."
                    end
                end
                -- gear that only exists in the backpack screen (Gorilla Gel and similar)
                if AutoEquipEnabled and (opened or autoOpen) then
                    local got = autoFix.guiGear()
                    if got then extra[#extra + 1] = "Equipped your " .. got .. "." end
                end
                -- say so when slot 1 got a stand-in because there is no Taser
                for _, g in ipairs(moved) do
                    if g.slot == "1" and PLAN.slot1Fallback then
                        extra[#extra + 1] = "No Taser found, so " .. PLAN.slot1Fallback.Name .. " went into slot 1."
                    end
                end
                -- anything still not right: say why, so it can be reported and fixed
                for _, g in ipairs(glows) do
                    if not equipOnly and needsFix(g) and g.status ~= "ok" then
                        local nm = (g.label:gsub("^.-· ", ""))
                        if (autoFix.tries[g.slot] or 0) == 0 then
                            failed[#failed + 1] = "Couldn't see " .. nm .. " in your backpack view."
                        else
                            failed[#failed + 1] = nm .. " didn't move after " .. autoFix.tries[g.slot] .. " methods."
                        end
                    end
                end
            end)
            SynthDragging = false
            if weOpened then pcall(closeInventory) end -- only close it if the script was the one that opened it
            autoFix.silent = false
            autoFix.busy = false
            if handsFree and #moved == 0 and #extra == 0 then
                autoFix.cool = os.clock() + 180 -- nothing to do or nothing worked: don't pop up again for a while
            end
            if #moved > 0 or #extra > 0 then
                autoFix.closedCycles = 0 -- it's working, keep going next time
                if autoFixNotify then pcall(autoFixNotify, moved, extra) end
            elseif #failed > 0 and handsFree then
                for _, f in ipairs(failed) do afLog(f) end
                local sig = table.concat(failed, "|")
                if sig ~= autoFix.lastFailSig or os.clock() - autoFix.lastFailAt > 300 then
                    autoFix.lastFailSig, autoFix.lastFailAt = sig, os.clock()
                    if autoFixFail then pcall(autoFixFail, failed) end
                end
            end
        end

        local function autoFixTick(now)
            if not AutoFixEnabled or autoFix.busy then return end

            if now >= autoFix.rearmAt then -- check again periodically, even after earlier give-ups
                autoFix.rearmAt = now + 90
                autoFix.closedCycles = 0
                autoFix.cool = math.min(autoFix.cool, now)
            end

            if autoOpen then -- backpack is open (you opened it)
                if autoFix.openSince == 0 then autoFix.openSince = now end
                if now < autoFix.nextTick then return end
                autoFix.nextTick = now + 0.15
                if now - autoFix.openSince < 0.15 then return end -- a brief moment to finish opening, then go
                if os.clock() - LastUserInput < 0.3 or Engine.run ~= nil then return end
                if os.clock() - LastCombat < 3 then return end -- mid-fight: never touch the backpack
                for _, g in ipairs(glows) do
                    if needsFix(g) and (autoFix.tries[g.slot] or 0) < MAXTRIES then
                        task.spawn(runAutoFix, false)
                        return
                    end
                end
                if brassNeeded() or planEvict() or planEquipEvict() then task.spawn(runAutoFix, false) end
                return
            end

            -- backpack is closed
            autoFix.openSince = 0
            autoFix.tries = {}
            if now < autoFix.nextClosed then return end
            if now < autoFix.cool or autoFix.closedCycles >= 3 or HighPing then
                autoFix.nextClosed = now + 1.5
                return
            end
            -- right after the script starts or you respawn it only needs a short pause; otherwise it waits for a quiet moment
            local idleNeeded = (now < autoFix.relaxUntil) and 1.0 or 3
            local busy = os.clock() - LastUserInput < idleNeeded or Engine.run ~= nil or os.clock() - LastCombat < 6
            local forced = autoFix.forced()
            autoFix.nextClosed = now + ((busy and not forced) and 6 or 1.5)
            rescanAll()
            local need = false
            for _, g in ipairs(glows) do
                if needsFix(g) then need = true break end
            end
            if brassNeeded() or planEvict() or planEquipEvict() then need = true end
            if busy and not forced then
                -- you're busy: ask instead of interrupting (tap the prompt to fix now, or it waits for a quiet moment)
                autoFix.badStreak = need and (autoFix.badStreak + 1) or 0
                if need and autoFix.badStreak >= 2 and now - autoFix.lastPrompt > 120 then
                    autoFix.lastPrompt = now
                    if slotGlow.promptFix then pcall(slotGlow.promptFix) end
                end
                return
            end
            autoFix.badStreak = need and (autoFix.badStreak + 1) or 0
            if autoFix.badStreak >= 2 or (forced and need) then -- the same problem twice in a row, so it isn't a blip
                autoFix.badStreak = 0
                task.spawn(runAutoFix, true)
            end
        end


        -- right after a spawn (and when the script first loads): wait for your tools to arrive, then open the backpack,
        -- equip your knuckles/gear, and close it. This ignores idle and combat checks on purpose.
        -- wait until the new character is fully there: alive, appearance loaded, your hotbar drawn, and your
        -- tools finished arriving (their count stable for a moment). Starting earlier is why it used to fail.
        local function waitForSpawnReady()
            local t0 = os.clock()
            while os.clock() - t0 < 10 do
                local ch = p.Character
                local hum = ch and ch:FindFirstChildOfClass("Humanoid")
                if hum and hum.Health > 0 and ch:FindFirstChild("HumanoidRootPart") then
                    local okA, loaded = pcall(function() return p:HasAppearanceLoaded() end)
                    if (not okA) or loaded then break end
                end
                task.wait(0.1)
            end
            local last, since = -1, os.clock()
            while os.clock() - t0 < 14 do
                pcall(resolveSlotFrames)
                local bp = p:FindFirstChildOfClass("Backpack")
                local n = (bp and #bp:GetChildren() or 0)
                if p.Character then
                    for _, c in ipairs(p.Character:GetChildren()) do if c:IsA("Tool") then n = n + 1 end end
                end
                if n ~= last then last, since = n, os.clock() end
                local barUp = rowRef and rowRef.cell and rowRef.cell.Parent and chainShown(rowRef.cell.Parent)
                if os.clock() - since > 0.5 and barUp then break end
                task.wait(0.1)
            end
            task.wait(0.25)
        end

        -- right after a spawn (and when the script first loads): wait for the spawn to finish, then open the
        -- backpack, equip your knuckles/gear, and close it. Ignores idle and combat checks on purpose.
        local function respawnEquip()
            if not AutoEquipEnabled then return end
            waitForSpawnReady()
            for attempt = 1, 2 do
                pcall(scanInventory)
                if #equipQueue() == 0 and not planEquipEvict() and not AutoFixMemory.guiGear then return end
                local w0 = os.clock()
                while autoFix.busy and os.clock() - w0 < 10 do task.wait(0.1) end
                if autoFix.busy then return end
                autoFix.cool = 0
                runAutoFix(true, true)
                if #equipQueue() == 0 and not planEquipEvict() then return end
                task.wait(1.5) -- one more go after a short pause if anything is still not on
                autoFix.equipTries = {}
            end
        end
        task.spawn(respawnEquip)

        -- run a check shortly after the script starts, and every time you respawn
        autoFix.relaxUntil = os.clock() + 30
        autoFix.nextClosed = os.clock() + 5
        -- notice fights: taking damage or swinging a tool counts as combat
        local function hookCombat(char)
            task.spawn(function()
                local hum = char:WaitForChild("Humanoid", 8)
                if hum then
                    local last = hum.Health
                    hum.HealthChanged:Connect(function(h)
                        if h < last then LastCombat = os.clock() end
                        last = h
                    end)
                end
            end)
            local function hookTool(t)
                if t:IsA("Tool") then t.Activated:Connect(function() LastCombat = os.clock() end) end
            end
            for _, t in ipairs(char:GetChildren()) do hookTool(t) end
            char.ChildAdded:Connect(hookTool)
        end
        if p.Character then hookCombat(p.Character) end
        p.CharacterAdded:Connect(hookCombat)
        local bp = p:FindFirstChildOfClass("Backpack")
        if bp then
            -- a tool just appeared in your backpack (bought, found, given): re-plan and check again right away,
            -- so a new Taser or a better weapon is put where it belongs as soon as it shows up
            bp.ChildAdded:Connect(function(t)
                if not t:IsA("Tool") then return end
                autoFix.nextClosed = 0
                autoFix.cool = 0
                autoFix.closedCycles = 0
                autoFix.badStreak = 1
                autoFix.relaxUntil = os.clock() + 20
            end)
            local function hookBp(t)
                if t:IsA("Tool") then t.Activated:Connect(function() LastCombat = os.clock() end) end
            end
            for _, t in ipairs(bp:GetChildren()) do hookBp(t) end
            bp.ChildAdded:Connect(hookBp)
        end

        p.CharacterAdded:Connect(function()
            autoFix.equipDone = {}
            autoFix.equipTries = {}
            autoFix.spawnAt = os.clock()
            task.spawn(respawnEquip)
            autoFix.cool = 0
            autoFix.closedCycles = 0
            autoFix.badStreak = 1
            autoFix.nextClosed = os.clock() + 5
            autoFix.relaxUntil = os.clock() + 30
        end)

        -- Calm looping tone for as long as the backpack is open. Its volume swells and dips with
        -- the SAME pulse as the glow, so the sound and the flashing slots breathe together.
        -- Fades in when the backpack opens, fades out when it closes. Respects the "Sounds" switch.
        local INV_VOLUME = 0.25 -- loudest point of each pulse; lower = quieter, higher = louder
        local INV_FLOOR = 0.15  -- how quiet it gets between pulses (0 = fully silent, 1 = no pulsing)
        local invLoop, invGain, invWant, invLoading = nil, 0, false, false

        local function stopInventoryLoop()
            invWant = false -- updateInventoryLoop fades it out and cleans it up
        end

        local function startInventoryLoop()
            if not SoundsEnabled then return end
            if invLoop then invWant = true return end -- still fading out from before: just bring it back
            if invLoading then return end
            invLoading = true
            task.spawn(function()
                local id = loadSoundAsset("invhum")
                invLoading = false
                if not id or invLoop or not autoOpen then return end -- it may have closed while loading
                pcall(function()
                    local s = Instance.new("Sound")
                    s.Name = "ExoticaInventoryAmbience"
                    s.SoundId = id
                    s.Looped = true
                    s.Volume = 0
                    s.Parent = SoundService
                    s:Play()
                    invGain, invWant, invLoop = 0, true, s
                end)
            end)
        end

        -- runs every frame: fade in/out + follow the glow's pulse
        local function updateInventoryLoop(dt, now)
            local s = invLoop
            if not s then return end
            if invWant then
                invGain = math.min(1, invGain + dt / 2.0)   -- ~2s fade in
            else
                invGain = math.max(0, invGain - dt / 0.6)   -- ~0.6s fade out
                if invGain <= 0.001 then
                    invLoop = nil
                    pcall(function() s:Destroy() end)
                    return
                end
            end
            local pulse = 0.5 + 0.5 * math.sin(now * 4.5)   -- identical to the glow's pulse
            pcall(function() s.Volume = INV_VOLUME * invGain * (INV_FLOOR + (1 - INV_FLOOR) * pulse) end)
        end

        local expiry, pinned, alpha, nextCheck, nextResolve, nextWatch, nextIdleScan, nextGuiScan = 0, false, 0, 0, 0, 0, 0, 0
        -- true when something is out of place (re-reads your hotbar first); nil if it can't tell yet
        function slotGlow.anythingToFix()
            if #PLAN.allTools == 0 then pcall(scanInventory) end
            if #PLAN.allTools == 0 then return nil end
            rescanAll()
            for _, g in ipairs(glows) do
                if needsFix(g) then return true end
            end
            return (brassNeeded() or planEvict() ~= nil or planEquipEvict() ~= nil)
        end
        -- which hotbar slot a named weapon is in right now (nil if it isn't on the hotbar)
        function slotGlow.slotOfName(name)
            for _, tool in ipairs(PLAN.allTools) do
                if tool.Name == name then
                    for pos, c in pairs(hotCellsByPos) do
                        if c.Parent and cellMatchesTool(c, tool) then return pos end
                    end
                end
            end
            return nil
        end
        function slotGlow.refreshSlots()
            pcall(scanInventory)

        end
        -- your weapons, for the custom macro builder: { name, kind, slot }
        function slotGlow.ownedWeapons()
            local out = {}
            for _, t in ipairs(PLAN.owned) do
                local e = WEAPONS.tool(t)
                if e and (e.kind == "rpg" or e.kind == "taser" or e.kind == "dps" or e.kind == "acc" or e.kind == "fixed" or e.kind == "loveboard") then
                    out[#out + 1] = { name = t.Name, kind = e.kind, slot = slotGlow.slotOfName(t.Name) }
                end
            end
            table.sort(out, function(a, b) return a.name:lower() < b.name:lower() end)
            return out
        end

        -- what the notices use: which required weapons you're missing, and which of your picks are low grade
        function slotGlow.comboReport()
            local rep = { ready = false, missing = {}, low = {} }
            if #PLAN.allTools == 0 then return rep end
            rep.ready = true
            local need = {
                { "1", "a Taser (slot 1)", true }, { "4", "a high-DPS weapon for slot 4", true },
                { "5", "an RPG (slot 5)", true }, { "8", "your Fist with knuckles, or a high-DPS weapon (slot 8)", true },
                { "0", "a high-DPS weapon for slot 0", true },
                { "6", "the Loveboard (slot 6)", false }, { "7", "the Golden Snowball Launcher (slot 7)", false },
                { "9", "the Snowball Launcher (slot 9)", false },
            }
            for _, n in ipairs(need) do
                local have
                if n[3] then have = PLAN.slotTool[n[1]] ~= nil else have = expected[n[1]] and expected[n[1]].found end
                if not have then rep.missing[#rep.missing + 1] = n[2] end
            end
            local hasKnuckle = false
            for _, t in ipairs(PLAN.equipAll) do
                local e = WEAPONS.tool(t)
                if e and e.group == "knuckle" then hasKnuckle = true end
            end
            for _, slot in ipairs({ "4", "8", "0" }) do
                local t = PLAN.slotTool[slot]
                local e = t and WEAPONS.tool(t)
                if e and e.kind == "dps" and e.tier <= 50 then
                    rep.low[#rep.low + 1] = t.Name
                elseif e and e.kind == "fist" and not hasKnuckle then
                    rep.low[#rep.low + 1] = "Fist without knuckles"
                end
            end
            return rep
        end
        function slotGlow.flash(seconds) expiry = math.max(expiry, clock() + seconds) end
        function slotGlow.pin(on) pinned = on end

        slotGlow.conn = RunService.RenderStepped:Connect(function(dt)
            if not isCurrentInstance() then slotGlow.conn:Disconnect() return end -- an older copy must not keep running
            local now = clock()
            if now >= nextWatch then -- cheap: is the backpack/inventory open right now?
                nextWatch = now + 0.25
                local open = false
                if invWatchDefault and invWatchDefault.Parent and chainShown(invWatchDefault) then open = true end
                if invWatchGrid and invWatchGrid.Parent and chainShown(invWatchGrid) then open = true end
                if open ~= autoOpen then
                    autoOpen = open
                    if open then
                        if not autoFix.silent then startInventoryLoop() end
                    else
                        stopInventoryLoop()
                    end
                end
                if invLoop and not SoundsEnabled then stopInventoryLoop() end
            end
            updateInventoryLoop(dt, now)
            autoFixTick(now)
            local flashing = false
            for _, g in ipairs(glows) do
                if (g.flashUntil or 0) > now then flashing = true break end
            end
            local want = pinned or now < expiry or autoOpen or flashing
            local flashOnly = flashing and not (pinned or now < expiry or autoOpen)
            if not want and alpha < 0.01 and now >= nextIdleScan then
                nextIdleScan = now + 2.5 -- idle: keep an eye out for the inventory so opening it can auto-show the glow
                pcall(resolveSlotFrames)
            end
            alpha = alpha + ((want and 1 or 0) - alpha) * math.min(1, dt * 10)
            if alpha < 0.01 then
                if layer.Visible then layer.Visible = false end
                return
            end
            layer.Visible = true

            if clock() >= nextCheck then
                nextCheck = clock() + 0.25
                if clock() >= nextResolve then
                    nextResolve = clock() + 1.5
                    pcall(resolveSlotFrames)
                end
                pcall(scanGlobalTools)
                if clock() >= nextGuiScan then
                    nextGuiScan = clock() + 3
                    pcall(scanGuiNames)
                end
                pcall(scanInventory)
                for _, g in ipairs(glows) do pcall(checkSlot, g) end
            end

            local origin, screen = sg.AbsolutePosition, sg.AbsoluteSize
            local pulse = 0.5 + 0.5 * math.sin(clock() * 4.5)
            for _, g in ipairs(glows) do
                local pos, size = slotRect(g, origin, screen)
                local isFlash = (g.flashUntil or 0) > now
                if flashOnly and not isFlash then pos = nil end -- only the slots it just fixed light up
                g.box.Visible = pos ~= nil
                g.tag.Visible = pos ~= nil
                if not pos then continue end
                g.box.Position = UDim2.fromOffset(pos.X, pos.Y)
                g.box.Size = UDim2.fromOffset(size.X, size.Y)
                if isFlash then -- "just moved here": a faster, brighter green pulse
                    local fp = 0.5 + 0.5 * math.sin(clock() * 14)
                    g.box.BackgroundTransparency = 1 - alpha * (0.30 + 0.40 * fp)
                    g.edge.Transparency = 1 - alpha * (0.8 + 0.2 * fp)
                    g.haloEdge.Transparency = 1 - alpha * (0.5 + 0.5 * fp)
                else
                    g.box.BackgroundTransparency = 1 - alpha * (0.08 + 0.12 * pulse)
                    g.edge.Transparency = 1 - alpha * (0.6 + 0.4 * pulse)
                    g.haloEdge.Transparency = 1 - alpha * (0.2 + 0.35 * pulse)
                end
                -- stagger tags (alternating heights) so neighbouring slots don't overlap
                local lift = (g.pos % 2 == 0) and 8 or 26
                g.tag.Position = UDim2.fromOffset(pos.X + size.X / 2, pos.Y - lift)
                g.tag.BackgroundTransparency = 1 - alpha * 0.7
                g.tag.TextTransparency = 1 - alpha
            end
        end)
    end, function(err) warn("[Exotica] slot features failed to start:", tostring(err)) end)

    -- Auto Slot Fix: tells you what it detected and moved (one banner for the whole batch, with a sound)
    autoFixNotify = function(moved, extra)
        local lines = {}
        local combo = #moved > 0
        for _, e in ipairs(extra or {}) do
            if e:find("^Moved ") then combo = true end
        end
        if combo then lines[1] = "The best combo has been selected for you." end
        for _, g in ipairs(moved) do
            local item = (g.label:gsub("^.-· ", ""))
            lines[#lines + 1] = "Detected " .. item .. " and moved it to slot " .. g.slot .. "."
        end
        for _, e in ipairs(extra or {}) do lines[#lines + 1] = e end
        showNotification({
            title = combo and "Best Combo Selected" or "Gear Equipped",
            badge = "OK",
            badgeSize = 12,
            color = COLORS.green,
            width = 330,
            duration = 7 + 1.5 * #lines,
            hint = "Auto Slot Fix",
            sounds = combo and { "combo" } or { "chime" },
            items = lines,
        })
    end

    -- High-ping watch: when your connection lags, the aim assist and Auto Slot Fix get slower and less accurate,
    -- so you're told once (and the backpack visits pause until it recovers).
    task.spawn(function()
        local Stats = game:GetService("Stats")
        local samples, lastWarn = {}, -1000
        local function readPing()
            local ok, v = pcall(function() return Stats.Network.ServerStatsItem["Data Ping"]:GetValue() end)
            if ok and type(v) == "number" and v > 0 then return v end
            local ok2, v2 = pcall(function() return p:GetNetworkPing() * 2000 end)
            return (ok2 and type(v2) == "number") and v2 or 0
        end
        while true do
            task.wait(1)
            samples[#samples + 1] = readPing()
            if #samples > 5 then table.remove(samples, 1) end
            local sum = 0
            for _, x in ipairs(samples) do sum = sum + x end
            local avg = sum / #samples
            if not HighPing and #samples >= 4 and avg > 250 then
                HighPing = true
                if os.clock() - lastWarn > 90 then
                    lastWarn = os.clock()
                    pcall(showNotification, {
                        title = "High Ping Detected",
                        badge = "!",
                        color = COLORS.warn,
                        width = 330,
                        duration = 10,
                        hint = "Connection",
                        sounds = { "popup", "warning" },
                        items = {
                            string.format("Your ping is about %d ms.", math.floor(avg + 0.5)),
                            "While you're laggy, the aim assist and Auto Slot Fix can be slower and less accurate.",
                            "They work normally again as soon as your connection recovers.",
                        },
                    })
                end
            elseif HighPing and avg < 180 then
                HighPing = false
            end
        end
    end)

    ------------------------------------------------------------
    xpcall(function()
        local KEYS = { "One", "Two", "Three", "Four", "Five", "Six", "Seven", "Eight", "Nine", "Zero" }
        local HttpService = game:GetService("HttpService")
        local CM = { defs = {}, file = "ExoticaMacros_custom.json", page = "list", draft = nil, editing = nil, W = 330 }
        local capture = nil -- when set, the next key you press is handed to it

        local function say(title, body)

                    hint = "Tap to dismiss", duration = 4, sounds = {} })
            end

        local function saveDefs()
            AutoFixMemory.customMacros = CM.defs
            pcall(function() if writefile then writefile(CM.file, HttpService:JSONEncode(CM.defs)) end end)
        end
        local function loadDefs()
            local ok, data = pcall(function()
                if isfile and isfile(CM.file) then return HttpService:JSONDecode(readfile(CM.file)) end
                return AutoFixMemory.customMacros
            end)
            if ok and type(data) == "table" then CM.defs = data end
        end
        local function idOf(def) return "Custom: " .. def.name end

        -- turn a macro's steps into the same kind of timed events the built-in macros use
        local function buildEvents(def)
            slotGlow.refreshSlots()
            local seq = newSeq()
            local steps = def.steps
            local lastKind
            local i = 1
            while i <= #steps do
                local st = steps[i]
                local gap, j = 0, i + 1
                while steps[j] and steps[j].t == "wait" do
                    gap = gap + math.floor((steps[j].sec or 0.1) * 1000)
                    j = j + 1
                end
                if st.t == "weapon" then
                    local pos = slotGlow.slotOfName(st.name)
                    if pos and KC[KEYS[pos]] then seq.key(KC[KEYS[pos]], 10, gap + 30) end
                    lastKind = st.kind
                elseif st.t == "slot" then
                    if KC[KEYS[st.slot]] then seq.key(KC[KEYS[st.slot]], 10, gap + 30) end
                elseif st.t == "key" then
                    if KC[st.key] then seq.key(KC[st.key], 10, gap) end
                elseif st.t == "click" then
                    local tag
                    if st.aim then tag = (lastKind == "rpg") and "rpg" or "normal" end
                    seq.click(10, gap + 20, tag)
                end
                i = j
            end
            return seq.events
        end
        local function register(def)
            Macros[idOf(def)] = function() startSequence(buildEvents(def)) end
        end

        -- screen button for a custom macro (reuses the screen-button code)
        local function setButton(def, on)
            local id = idOf(def)
            if on then
                OSB.spec[id] = { icon = "🧩", label = def.name:sub(1, 8), color = COLORS.accent2, dx = def.bx or 0.8, dy = def.by or 0.72 }

                    def.bx, def.by = b.Position.X.Scale, b.Position.Y.Scale
                    b:Destroy()
                    OSB.buttons[id] = nil
                end
            end
        end

-- Exotica Hotbar Guard. Works as its own script, or paste the whole file into your script (before the final print).
-- It creates no top-level locals, so it cannot push your script over the 200-local limit.
-- Stops the game's anti-spam from switching your hotbar off and force-unequipping your tools.
-- ==========================================
-- HOTBAR GUARD (Baddies)
-- Cause found in the game's BackpackController: six Tool additions to your character with under 0.7s
-- between them trips an anti-spam that turns the Satchel hotbar off, force-unequips your tools and
-- waits 3 seconds. Fast macros trip it.
-- Fix 1: that anti-spam trigger is neutralised at the source (_disableSpam).
-- Fix 2: Satchel:SetBackpackEnabled(false) is blocked unless the game has a real reason (driving, duel,
--        arrested, frozen, dress room...), and a watchdog turns the hotbar back on if it is still off.
-- Logger: while DEBUG is true, everything that hides the hotbar is printed and saved to
--        Exotica_Hotbar_Debug.txt (with who called it), so it can be sent for diagnosis.
-- ==========================================
task.spawn(function()
    local p = game:GetService("Players").LocalPlayer
    local GUARD_ENABLED = true
    local DEBUG = false -- set true to print what it blocks/paces and save Exotica_Hotbar_Debug.txt

    -- only the newest copy of this guard stays active (running it twice used to make two keepers fight)
    local env = (getgenv and getgenv()) or _G
    env.__ExoticaGuardGen = (env.__ExoticaGuardGen or 0) + 1
    local myGen = env.__ExoticaGuardGen

    -- on/off switch: set getgenv().__ExoticaHotbarGuardOn = false to turn the guard off, true to turn it back on
    -- (the Exotica panel's "Hotbar Guard" switch does exactly this). Unset = on. When off, everything passes through.
    task.spawn(function()
        while true do
            task.wait(0.1)
            if env.__ExoticaGuardGen ~= myGen then GUARD_ENABLED = false return end
            GUARD_ENABLED = (env.__ExoticaHotbarGuardOn ~= false)
        end
    end)

    local RS = game:GetService("ReplicatedStorage")
    local t0 = os.clock()
    local logLines, flushQueued = {}, false
    local function dlog(...)
        if not DEBUG then return end
        local parts = { string.format("[%7.2f]", os.clock() - t0) }
        for _, v in ipairs({ ... }) do parts[#parts + 1] = tostring(v) end
        local line = table.concat(parts, " ")
        print("[Exotica HotbarGuard]", line)
        logLines[#logLines + 1] = line
        if #logLines > 400 then table.remove(logLines, 1) end
        if writefile and not flushQueued then
            flushQueued = true
            task.delay(1.5, function()
                flushQueued = false
                pcall(writefile, "Exotica_Hotbar_Debug.txt", table.concat(logLines, "\n"))
            end)
        end
    end

    local LEGIT_BLOCKS = {
        "Freezed", "BeingDogLeashed", "BeingTombstoneTrapped", "BeingArrested", "USER_DRIVING",
        "USER_VEHICLE_SITTING", "IN_DRESS_ROOM", "CARRYING_EGG", "UNDERGROUND_FIGHTING", "IsOnDuel",
        "BeingSchoolLockerTrapped", "IsJuggernautOfRunEvent", "GravityGunEffect", "ChampionRingQueue",
    }
    local function activeBlocks()
        local on = {}
        for _, a in ipairs(LEGIT_BLOCKS) do
            if p:GetAttribute(a) then on[#on + 1] = a end
        end
        return on
    end

    -- sets t[k] = v even if the table is read-only (frozen), when the executor allows it; logs if it can't
    local function patch(t, k, v)
        if pcall(function() t[k] = v end) then return true end
        pcall(function() if setreadonly then setreadonly(t, false) end end)
        local ok2, err = pcall(function() t[k] = v end)
        if ok2 then return true end
        dlog("could not patch", k, "|", tostring(err))
        return false
    end

    local Satchel, Ctrl
    pcall(function()
        local mods = RS:WaitForChild("Modules", 15)
        Satchel = require(mods:WaitForChild("Satchel", 15))
    end)
    pcall(function()
        local scripts = p:WaitForChild("PlayerScripts", 15)
        local controllers = scripts:WaitForChild("Controllers", 30)
        Ctrl = require(controllers:WaitForChild("BackpackController", 30))
    end)
    dlog("Satchel:", Satchel and "found" or "MISSING", "| BackpackController:", type(Ctrl) == "table" and "found" or "MISSING")

    -- Fix 1: the anti-spam shutdown never runs (no hotbar off, no forced unequip)
    if type(Ctrl) == "table" then
        local origSpam = Ctrl._disableSpam
        patch(Ctrl, "_disableSpam", function(self, ...)
            if not GUARD_ENABLED then return origSpam(self, ...) end
            dlog("BLOCKED the game's anti-spam hotbar shutdown")
        end)
    end

    -- Fix 2: Satchel can't be switched off without a real reason; every attempt is logged with its caller
    if Satchel then
        local origSet = Satchel.SetBackpackEnabled
        patch(Satchel, "SetBackpackEnabled", function(self, value, ...)
            if value == false then
                local why = activeBlocks()
                dlog("SetBackpackEnabled(false) | game reasons:", #why > 0 and table.concat(why, ",") or "NONE")
                dlog((debug.traceback("caller", 2):gsub("\n", " <- ")))
                if GUARD_ENABLED and #why == 0 then
                    dlog("-> ignored")
                    return
                end
            end
            return origSet(self, value, ...)
        end)
    end

    -- logger: tool add / remove timing on your character (shows the burst that trips the game)
    local function watchChar(char)
        if not char then return end
        char.ChildAdded:Connect(function(c) if c:IsA("Tool") then dlog("equipped", c.Name) end end)
        char.ChildRemoved:Connect(function(c) if c:IsA("Tool") then dlog("unequipped", c.Name) end end)
    end
    watchChar(p.Character)
    p.CharacterAdded:Connect(watchChar)

    -- ==== TOOL KEEPER ====
    -- If a tool is pulled out of your hands right after you equipped it, and nothing on your side asked for that
    -- (not the hotbar, not your macros) and the game has no real reason (stunned, ragdolled, arrested, in a duel...),
    -- the tool is put back in your hand. Works whether the client or the server did it.
    local KEEP_TOOLS = false -- off by default: pacing (below) prevents the pull-outs, and re-equipping can add noise. Set true to bring it back.
    -- Why it waits: the game punishes a burst of equips. If the tool goes straight back into your hand, that
    -- is one more equip in the burst, and it gets pulled out again (the loop in the log). Waiting until the
    -- burst window (0.7s) has passed lets the counters reset, so the tool stays. Set to 0 for instant.
    local KEEPER_WAIT = 0.75
    local lastToolAdd = -10

    -- ==== EQUIP PACING ====
    -- The log shows that once the game decides you are spamming, EVERY tool you equip is pulled out again about
    -- 60-80ms later, even after waiting 1.4s. So the best fix is not to trip it at all: the game trips on the
    -- 6th equip in a row with under 0.7s between them. This holds the 6th equip back until a 0.7s quiet gap has
    -- reset its counter, then equips it. (Costs ~0.7s on that one equip, instead of losing the weapon.)
    local PACE_EQUIPS = true -- ON: with it off, about 4% of equips (the 6th in a burst, usually Loveboard) still got pulled out. Set false to remove the ~0.7s pause on that 6th equip.
    local PACE_MAX = 5      -- quick equips allowed in a row. If it still trips, try 4 or 3.
    local PACE_GAP = 0.72   -- the game resets its counter after a 0.7s gap
    local burst = 0
    local paceBypass = false
    local paceToken = 0
    local lastEquipped = nil
    local lastUnequipAsk = -10 -- last time anything on your client called Humanoid:UnequipTools
    pcall(function()
        if hookmetamethod and getnamecallmethod then
            local oldNC
            oldNC = hookmetamethod(game, "__namecall", (newcclosure or function(f) return f end)(function(self, ...)
                local m = getnamecallmethod()
                if m == "UnequipTools" then
                    lastUnequipAsk = os.clock()
                elseif m == "EquipTool" and PACE_EQUIPS and GUARD_ENABLED and not paceBypass then
                    local tool = ...
                    local now = os.clock()
                    local okOwn = pcall(function() return self.Parent == p.Character end)
                    if okOwn and self.Parent == p.Character and burst >= PACE_MAX and now - lastToolAdd < 0.7 then
                        -- this would be the 6th quick equip: hold it back until the counter has reset
                        paceToken = paceToken + 1
                        local myToken = paceToken
                        local hum = self
                        local delay = (lastToolAdd + PACE_GAP) - now
                        task.delay(delay, function()
                            if myToken ~= paceToken then return end -- a newer equip replaced this one
                            paceBypass = true
                            pcall(function() hum:EquipTool(tool) end)
                            paceBypass = false
                        end)
                        dlog("PACED an equip of", tostring(tool), "by", math.floor(delay * 1000), "ms (it would have been equip #" .. (burst + 1) .. " in a row)")
                        return
                    end
                end
                return oldNC(self, ...)
            end))
        else
            dlog("no hookmetamethod: the keeper can't tell your own unequips apart, it only reacts within 0.6s of an equip")
        end
    end)

    local equippedAt = setmetatable({}, { __mode = "k" })
    local restoreLog = setmetatable({}, { __mode = "k" })
    local keepState = setmetatable({}, { __mode = "k" }) -- per tool: how many quick put-backs got pulled out again

    local function safeToRestore(char)
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        if not hum or hum.Health <= 0 then return false, nil end
        for _, n in ipairs({ "RagdollTrigger", "Dead", "BeingCarried", "stunned" }) do
            if char:FindFirstChild(n) then return false, nil end
        end
        if hum:GetAttribute("Ragdoll") or p:GetAttribute("InDanceBombEffect") or p:GetAttribute("InPotionBubble") then
            return false, nil
        end
        if #activeBlocks() > 0 then return false, nil end
        return true, hum
    end

    local function handHasTool(char)
        for _, c in ipairs(char:GetChildren()) do
            if c:IsA("Tool") then return true end
        end
        return false
    end

    local function watchKeeper(char)
        if not char then return end
        char.ChildAdded:Connect(function(c)
            if c:IsA("Tool") then
                local nowA = os.clock()
                if nowA - lastToolAdd >= 0.7 then burst = 0 end -- same rule the game uses
                burst = burst + 1
                equippedAt[c] = nowA
                lastToolAdd = nowA
                lastEquipped = c
            end
        end)
        char.ChildRemoved:Connect(function(c)
            if not (KEEP_TOOLS and GUARD_ENABLED) or not c:IsA("Tool") then return end
            local t = equippedAt[c]
            if not t then return end
            local now = os.clock()
            if now - t > 0.6 then return end                -- only a tool that was JUST equipped
            if lastUnequipAsk > t + 0.002 then return end   -- unequipped on purpose by you / the hotbar / your macro
            -- first try is quick. If the tool gets pulled out AGAIN right after a put-back, the quick way is not
            -- working, so each further try waits longer (0.06s, then KEEPER_WAIT, then twice that).
            local st = keepState[c] or { n = 0, at = -10 }
            keepState[c] = st
            if now - st.at < 0.4 then st.n = st.n + 1 else st.n = 0 end
            local ladder = { 0.06, KEEPER_WAIT, KEEPER_WAIT * 2 }
            local want = ladder[math.min(st.n + 1, #ladder)]
            local waitFor = math.max(0.06, (lastToolAdd + want) - now)
            task.delay(waitFor, function()
                local ok, hum = safeToRestore(char)
                if not ok then return end
                if handHasTool(char) then return end        -- you already picked something else
                if lastEquipped ~= c then return end        -- you equipped a different tool since; do not bring this one back
                local bp = p:FindFirstChildOfClass("Backpack")
                if not bp or c.Parent ~= bp then return end
                -- no fighting: at most 3 puts-back per tool every 2 seconds
                local log2 = restoreLog[c] or {}
                for i = #log2, 1, -1 do
                    if os.clock() - log2[i] > 2 then table.remove(log2, i) end
                end
                if #log2 >= 3 then return end
                log2[#log2 + 1] = os.clock()
                restoreLog[c] = log2
                paceBypass = true
                pcall(function() hum:EquipTool(c) end)
                paceBypass = false
                st.at = os.clock()
                dlog("KEEPER put", c.Name, "back in your hand after waiting", math.floor(waitFor * 1000), "ms (it was pulled out", math.floor((now - t) * 1000), "ms after you equipped it)")
            end)
        end)
    end
    watchKeeper(p.Character)
    p.CharacterAdded:Connect(watchKeeper)

    -- watchdog: if the hotbar is off with no game reason, turn it back on
    local lastState
    while true do
        task.wait(0.1)
        if env.__ExoticaGuardGen ~= myGen then
            GUARD_ENABLED = false -- a newer copy took over; this one goes inert
            return
        end
        if Satchel then
            local ok, enabled = pcall(function() return Satchel:GetBackpackEnabled() end)
            if ok and enabled ~= lastState then
                lastState = enabled
                dlog("hotbar enabled =", tostring(enabled))
            end
            if GUARD_ENABLED and ok and enabled == false and #activeBlocks() == 0 then
                local hum = p.Character and p.Character:FindFirstChildOfClass("Humanoid")
                if hum and hum.Health > 0 then
                    pcall(function()
                        Satchel:SetBackpackEnabled(true)
                    end)
                    if type(Ctrl) == "table" then pcall(function() Ctrl._enabled = true Ctrl._spamDelay = false end) end
                    dlog("watchdog turned the hotbar back on")
                end
            end
        end
    end
end)

print("Exotica: Master Suite Initialized Successfully!")
